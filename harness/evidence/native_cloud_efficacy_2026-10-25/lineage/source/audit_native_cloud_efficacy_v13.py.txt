def sha(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(8 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()


def same_bits(left, right) -> bool:
    a, b = np.asarray(left), np.asarray(right)
    return a.shape == b.shape and a.dtype == b.dtype and a.tobytes() == b.tobytes()


def _write_json_new(path: Path, payload: dict) -> str:
    encoded = (json.dumps(payload, sort_keys=True, indent=2, allow_nan=False) + "\n").encode()
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
    with os.fdopen(fd, "wb") as stream:
        stream.write(encoded)
        stream.flush()
        os.fsync(stream.fileno())
    return hashlib.sha256(encoded).hexdigest()


def _write_npz_new(path: Path, arrays: dict) -> dict:
    with Path(path).open("xb") as stream:
        np.savez_compressed(stream, **arrays)
        stream.flush()
        os.fsync(stream.fileno())
    size = Path(path).stat().st_size
    if size > 50 * 1024 * 1024:
        raise ValueError(f"split native bundle exceeds 50 MiB: {path} ({size} bytes)")
    return {
        "path": str(path), "sha256": sha(path), "bytes": size,
        "keys": sorted(arrays),
        "arrays": {name: {"shape": list(np.asarray(value).shape), "dtype": np.asarray(value).dtype.str}
                   for name, value in arrays.items()},
    }


def _hash_one(item):
    label, path, expected = item
    path = Path(path)
    if not path.is_file() or path.is_symlink():
        return label, str(path), False, "missing_or_symlink"
    actual = sha(path)
    return label, str(path), actual == expected, actual


def _quantiles(values):
    a = np.asarray(values, dtype=np.float64)
    if not a.size:
        return {"count": 0}
    q = np.quantile(a, [0.0, 0.5, 0.9, 0.95, 0.99, 0.999, 1.0])
    return {
        "count": int(a.size), "mean": float(np.mean(a)), "median": float(q[1]),
        "p90": float(q[2]), "p95": float(q[3]), "p99": float(q[4]),
        "p99_9": float(q[5]), "max": float(q[6]),
        "rms_magnitude": float(np.sqrt(np.mean(a * a))),
    }


def _bt_delta_summary(delta_bt_k):
    delta = np.asarray(delta_bt_k, dtype=np.float64)
    if not delta.size:
        return {"count": 0}
    absolute = np.abs(delta)
    q = np.quantile(absolute, [0.0, 0.5, 0.9, 0.95, 0.99, 0.999, 1.0])
    return {
        "count": int(delta.size),
        "absolute_delta_quantiles_K": {
            "min": float(q[0]), "p50": float(q[1]), "p90": float(q[2]),
            "p95": float(q[3]), "p99": float(q[4]), "p99_9": float(q[5]), "max": float(q[6]),
        },
        "rms_increment_K": float(np.sqrt(np.mean(delta * delta))),
        "count_abs_ge_0p1K": int(np.count_nonzero(absolute >= 0.1)),
        "count_abs_ge_1K": int(np.count_nonzero(absolute >= 1.0)),
        "count_abs_lt_0p1K": int(np.count_nonzero(absolute < 0.1)),
        "signed_mean_delta_K": float(np.mean(delta)),
    }


def _residual_stats(residual):
    residual = np.asarray(residual, dtype=np.float64)
    if not residual.size:
        return {"count": 0}
    absolute = np.abs(residual)
    return {
        "count": int(residual.size),
        "bias_mean_K": float(np.mean(residual)),
        "rmse_K": float(np.sqrt(np.mean(residual * residual))),
        "mae_K": float(np.mean(absolute)),
        "median_abs_residual_K": float(np.median(absolute)),
        "p90_abs_residual_K": float(np.quantile(absolute, 0.9)),
        "p95_abs_residual_K": float(np.quantile(absolute, 0.95)),
        "p99_abs_residual_K": float(np.quantile(absolute, 0.99)),
        "max_abs_residual_K": float(np.max(absolute)),
    }


def _residual_skill(baseline_bt, candidate_bt, observed_bt, selector):
    b = np.asarray(baseline_bt, dtype=np.float64)[selector]
    c = np.asarray(candidate_bt, dtype=np.float64)[selector]
    o = np.asarray(observed_bt, dtype=np.float64)[selector]
    if not b.size:
        return {"support_count": 0}
    if not (np.isfinite(b).all() and np.isfinite(c).all() and np.isfinite(o).all()):
        raise ValueError("nonfinite BT on fixed residual cohort")
    br = b - o
    cr = c - o
    absolute_gain = np.abs(br) - np.abs(cr)  # positive means candidate closer to observation
    return {
        "support_count": int(b.size),
        "baseline_B_minus_O_residual": _residual_stats(br),
        "baseline_O_minus_B_residual": _residual_stats(-br),
        "candidate_A_minus_O_residual": _residual_stats(cr),
        "candidate_O_minus_A_residual": _residual_stats(-cr),
        "baseline_minus_candidate_RMSE_improvement_K": float(np.sqrt(np.mean(br * br)) - np.sqrt(np.mean(cr * cr))),
        "baseline_minus_candidate_MAE_improvement_K": float(np.mean(np.abs(br)) - np.mean(np.abs(cr))),
        "absolute_residual_gain_baseline_minus_candidate_K": {
            "mean": float(np.mean(absolute_gain)),
            "median": float(np.median(absolute_gain)),
            "positive_count": int(np.count_nonzero(absolute_gain > 0.0)),
            "worsened_count": int(np.count_nonzero(absolute_gain < 0.0)),
            "unchanged_count": int(np.count_nonzero(absolute_gain == 0.0)),
            "improved_by_at_least_0p1K": int(np.count_nonzero(absolute_gain >= 0.1)),
            "improved_by_at_least_1K": int(np.count_nonzero(absolute_gain >= 1.0)),
        },
    }


def _phase_delta_stats(base, candidate, selector):
    b = np.asarray(base, dtype=np.float64)[selector]
    c = np.asarray(candidate, dtype=np.float64)[selector]
    d = c - b
    if not d.size:
        return {"columns": 0}
    return {
        "columns": int(d.size),
        "baseline": _quantiles(b),
        "candidate": _quantiles(c),
        "delta_candidate_minus_baseline": {
            **_quantiles(d),
            "positive_columns": int(np.count_nonzero(d > 0.0)),
            "negative_columns": int(np.count_nonzero(d < 0.0)),
            "zero_columns": int(np.count_nonzero(d == 0.0)),
        },
    }


def _level_stats(delta, selector):
    values = np.asarray(delta, dtype=np.float64)[selector]
    if not values.size:
        return {"columns": 0}
    q10, med, q90 = np.quantile(values, [0.1, 0.5, 0.9], axis=0)
    return {
        "columns": int(values.shape[0]),
        "native_k_bottomfirst_1based": list(range(1, values.shape[1] + 1)),
        "mean": np.mean(values, axis=0).tolist(),
        "median": med.tolist(),
        "p10": q10.tolist(),
        "p90": q90.tolist(),
        "max_abs": np.max(np.abs(values), axis=0).tolist(),
        "nonzero_columns_by_level": np.count_nonzero(values != 0.0, axis=0).tolist(),
        "positive_cells_by_level": np.count_nonzero(values > 0.0, axis=0).tolist(),
        "negative_cells_by_level": np.count_nonzero(values < 0.0, axis=0).tolist(),
    }
