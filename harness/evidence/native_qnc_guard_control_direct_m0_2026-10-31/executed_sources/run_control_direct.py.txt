def sha(p):
 h=hashlib.sha256()
 with Path(p).open('rb') as f:
  for b in iter(lambda:f.read(8<<20),b''): h.update(b)
 return h.hexdigest()

def source_tree_pin():
 """Pin operational harness/oracle code and bundled observation data."""
 files=sorted([*REPO.glob('harness/*.py'),*REPO.glob('oracle/**/*.py'),
               *REPO.glob('oracle/kdm6/obs/data/**/*')])
 files=[p for p in files if p.is_file() and '__pycache__' not in p.parts]
 h=hashlib.sha256()
 for p in files:
  h.update(p.relative_to(REPO).as_posix().encode()+b'\0'+sha(p).encode()+b'\n')
 return len(files),h.hexdigest()
