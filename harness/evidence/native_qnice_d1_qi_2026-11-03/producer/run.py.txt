#!/usr/bin/env python3
"""Isolated same-input D1 QI capture using the retained native host case."""

from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path
import shutil
import sys


ROOT = Path(__file__).resolve().parent
CANONICAL = Path("/Users/yhlee/KDM6AD-k")
PLAN = CANONICAL / "host/research_evidence/native_qnc_flux_trace_20261026/plans/pr457_v5/experiment_plan.json"
PLAN_SHA = "6032da73badc756c06410669d540daf53491afaa4f9705765ba7f270ac4343f7"
EXE = ROOT / "host_bundle/host/KIM-meso_v1.0/main/wrf.exe"
DSO = ROOT / "host_bundle/libtorch/install/lib/libkdm6_c.2.0.0.dylib"
WORKTREE = Path("/private/tmp/KDM6AD-qnice-pre-c3-audit")


def sha(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(8 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("mode", choices=("off", "on"))
    args = parser.parse_args()
    if sha(PLAN) != PLAN_SHA:
        raise ValueError("input plan changed")
    if shutil.disk_usage(ROOT).free < 25 * (1 << 30):
        raise OSError("less than 25 GiB free")
    plan = json.loads(PLAN.read_bytes())["runtime"]
    for path, expected in [(plan["origin"]["checkpoint_path"], plan["origin"]["checkpoint_sha256"]),
                           (plan["namelist_source_path"], plan["namelist_source_sha256"]),
                           *((a["path"], a["sha256"]) for a in plan["assets"])]:
        if sha(Path(path)) != expected:
            raise ValueError("input changed: " + path)
    run = ROOT / "runs" / args.mode
    run.mkdir(parents=True, exist_ok=False)
    (run / "scratch").mkdir()
    capture = run / "capture"
    if args.mode == "on":
        capture.mkdir()
    sys.path.insert(0, str(WORKTREE))
    from harness.native_host_transition import NativeCheckpoint, NativeInputs, PinnedFile
    from harness.native_live_stepper import NativeLiveStepper

    origin = NativeCheckpoint.capture(
        plan["origin"]["checkpoint_path"], expected_time=plan["start_time"],
        expected_itimestep=plan["origin"]["itimestep"])
    env = dict(plan["environment_without_trace"])
    env["KDM6AD_QNC_LIVE_GUARD"] = "1"
    if args.mode == "on":
        env.update(
            KDM6_NATIVE_CAPTURE_DIR=str(capture.resolve()),
            KDM6_NATIVE_CAPTURE_CALL="2",
            KDM6_NATIVE_CAPTURE_B="22617",
            KDM6_NATIVE_CAPTURE_ALL_K="1",
            KDM6_NATIVE_CAPTURE_LEGACY_SCALAR="0",
            KDM6_NATIVE_CAPTURE_MAX_ROWS="7000",
            KDM6_NATIVE_CAPTURE_MAX_BYTES=str(2 << 20),
            KDM6AD_NATIVE_QNICE_TRACE=str((capture / "host_qnice_pre_call.csv").resolve()),
        )
    external = NativeInputs(
        namelist=PinnedFile.bind(plan["namelist_source_path"]),
        assets=tuple((a["case_basename"], PinnedFile.bind(a["path"])) for a in plan["assets"]),
        environment=tuple(sorted(env.items())),
    )
    with NativeLiveStepper(
        initial=origin, external=external,
        executable=PinnedFile.bind(EXE), library=PinnedFile.bind(DSO),
        runner=PinnedFile.bind(WORKTREE / "harness/run_ss_case.py"),
        work_root=run / "scratch", steps=1, dt_seconds=20,
        trace_column=(164, 242), radiation_every_step=True,
        control_family="ANALYSIS_TQ", capture_ready_checkpoint=True,
    ) as stepper:
        ready = stepper.current
        if ready is None or ready.ready is None:
            raise RuntimeError("no READY state")
        final = stepper.advance(ready, ())
        result = {
            "schema": "kdm6.native_d1_qi_capture_run.v1",
            "mode": args.mode,
            "origin_sha256": origin.file.sha256,
            "startup_sha256": ready.checkpoint.file.sha256,
            "endpoint_sha256": final.checkpoint.file.sha256,
            "endpoint_path": str(final.checkpoint.file.path),
            "executable_sha256": sha(EXE),
            "dso_sha256": sha(DSO),
            "source_sha256": {name: sha(ROOT / "source/src" / name)
                              for name in ("coordinator.cpp", "runtime.cpp", "melt_freeze.cpp")},
            "object_sha256": {name: sha(ROOT / "build/CMakeFiles/kdm6.dir/src" / name)
                              for name in ("coordinator.cpp.o", "runtime.cpp.o", "melt_freeze.cpp.o")},
            "session_record_sha256": sha(stepper.case / "live_session.json"),
            "captures": {p.name: sha(p) for p in sorted(capture.iterdir())} if args.mode == "on" else {},
        }
        (run / "run_result.json").write_text(json.dumps(result, indent=2, sort_keys=True) + "\n")
        print(json.dumps(result, sort_keys=True))


if __name__ == "__main__":
    main()
