def value(row):
    bits = int(row["raw_hex"], 16)
    if row["dtype"] == "f32":
        require(bits < 1 << 32, "f32 bits")
        return struct.unpack("<f", bits.to_bytes(4, "little"))[0]
    if row["dtype"] == "f64":
        require(bits < 1 << 64, "f64 bits")
        return struct.unpack("<d", bits.to_bytes(8, "little"))[0]
    require(bits in (0, 1), "bool bits")
    return bool(bits)


def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def read_capture(path):
    require(path.suffix == ".csv" and not path.with_name(path.name + ".partial").exists(),
            "published capture file only")
    raw = path.read_bytes()
    text = raw.decode("ascii")
    lines = text.splitlines()
    require(len(lines) >= 3 and lines[0] == HEADER, "capture header")
    footer = FOOTER.fullmatch(lines[-1])
    require(footer is not None, "capture completion footer")
    rows = list(csv.DictReader(io.StringIO("\n".join(lines[:-1]) + "\n")))
    scalar = [row for row in rows if row["stage"] != "meta"]
    meta = [row for row in rows if row["stage"] == "meta"]
    require(len(scalar) == int(footer.group(1)) and len(meta) == int(footer.group(2)) == 1,
            "capture declared record count")
    keys = set()
    for row in rows:
        require(set(row) == set(HEADER.split(",")) and None not in row and
                row["dtype"] in {"f32", "f64", "bool"} and
                re.fullmatch(r"[0-9a-f]+", row["raw_hex"]) is not None,
                "capture row encoding")
        for name in ("one_step_call", "runtime_call", "substep", "b_zero", "k_zero", "B", "K"):
            require(row[name].isdigit(), "capture numeric index " + name)
        key = tuple(row[name] for name in
                    ("one_step_call", "runtime_call", "substep", "b_zero", "k_zero", "stage", "field"))
        require(key not in keys, "duplicate capture row")
        keys.add(key)
        require(row["stage"] in STAGES or (row["stage"] == "meta" and row["field"] == "dtcld"),
                "unknown capture stage/field")
        require(row["stage"] == "meta" or row["field"] in STAGES[row["stage"]],
                "capture field inventory")
        require(row["dtype"] == ("f32" if row["stage"] == "abi_ingress" or row["field"] == "fconst.pi_f32" else "bool" if row["field"] in {"icenuc.ifsat", "depsub.ifsat", "depsub.ice_complete_sublim"} else "f64"), "exact field dtype")
        require(row["dtype"] == "bool" if row["field"] in {
            "icenuc.ifsat", "depsub.ifsat", "depsub.ice_complete_sublim"} else True,
                "branch bool dtype")
        require(row["dtype"] == "f64" if row["stage"] == "meta" else True,
                "dtcld dtype")
        decoded = value(row)
        require(decoded == decoded and decoded not in (float("inf"), float("-inf")),
                "capture finite value")
    identities = {tuple(row[name] for name in ("one_step_call", "runtime_call", "substep", "B", "K"))
                  for row in rows}
    require(len(identities) == 1, "one invocation and shape per capture file")
    identity = next(iter(identities))
    b_count, level_count = map(int, identity[-2:])
    require(b_count > TARGET_B and level_count > max(TARGET_K), "selected tensor indices in bounds")
    expected_count = 12 * b_count + 2 * (sum(map(len, STAGES.values())) - len(STAGES["converted_ingress"]))
    require(len(scalar) == expected_count, "complete per-call scalar inventory")
    selected_map = {(int(row["k_zero"]), row["stage"]): set() for row in scalar
                    if int(row["b_zero"]) == TARGET_B and int(row["k_zero"]) in TARGET_K}
    for row in scalar:
        b, k = int(row["b_zero"]), int(row["k_zero"])
        require(0 <= b < b_count and 0 <= k < level_count, "capture index bounds")
        require(k in TARGET_K, "capture level inventory")
        require(row["stage"] == "converted_ingress" or b == TARGET_B,
                "non-target stage at unexpected batch")
        if b == TARGET_B:
            selected_map[(k, row["stage"])].add(row["field"])
    require(selected_map == {(k, stage): fields for k in TARGET_K
                             for stage, fields in STAGES.items()},
            "complete stage inventory for this call and each level")
    require(meta[0]["field"] == "dtcld" and value(meta[0]) > 0,
            "one positive dtcld metadata row")
    selected = [row for row in scalar if int(row["b_zero"]) == TARGET_B and
                int(row["k_zero"]) in TARGET_K]
    return {"sha256": sha(raw), "total_rows": len(rows), "identity": identity,
            "scalar_rows": len(scalar), "metadata_rows": len(meta),
            "selected": selected, "metadata": meta}


def unique_host_ingress_batch(path, host_records):
    """Find the unique B whose converted ingress matches six host words."""
    fields = {"scalar_ingress_state->qi": "QI",
              "scalar_ingress_forcing->p": "P",
              "scalar_ingress_forcing->den": "DEN"}
    expected = {(k - 1, name): float(host_records[("MP_INPUT", 3, 164, k, 242)].value(name))
                for k in (21, 23) for name in ("QI", "P", "DEN")}
    matches = {}
    with path.open("r", encoding="ascii") as stream:
        for row in csv.DictReader(line for line in stream if not line.startswith("#")):
            name = fields.get(row["field"])
            if row["stage"] != "converted_ingress" or name is None:
                continue
            k = int(row["k_zero"])
            if value(row) == expected[(k, name)]:
                matches.setdefault(int(row["b_zero"]), set()).add((k, name))
    exact = [b for b, fields_matched in matches.items() if len(fields_matched) == len(expected)]
    require(exact == [TARGET_B], "unique global-to-C++ converted-ingress batch")
    return {"batch_b_zero": TARGET_B, "six_host_fields_exact": True,
            "unique_candidate_count": len(exact)}
