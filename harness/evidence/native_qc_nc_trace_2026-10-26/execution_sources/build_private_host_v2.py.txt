def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            h.update(block)
    return h.hexdigest()


def run(argv, *, cwd=None, log=None, env=None) -> str:
    result = subprocess.run(
        [str(x) for x in argv], cwd=cwd, env=env, stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT, text=True, check=False,
    )
    if log is not None:
        Path(log).write_text(result.stdout)
    if result.returncode:
        raise subprocess.CalledProcessError(result.returncode, argv, output=result.stdout)
    return result.stdout


def clonefile(source: Path, destination: Path) -> None:
    """Create a copy-on-write APFS clone; fail closed if clonefile is unavailable."""
    if destination.exists() or destination.is_symlink():
        raise FileExistsError(destination)
    libc = ctypes.CDLL(None, use_errno=True)
    function = libc.clonefile
    function.argtypes = (ctypes.c_char_p, ctypes.c_char_p, ctypes.c_uint32)
    function.restype = ctypes.c_int
    status = function(os.fsencode(source), os.fsencode(destination), 0)
    if status != 0:
        error = ctypes.get_errno()
        raise OSError(error, os.strerror(error), str(destination))


def archive_members(path: Path) -> dict[tuple[str, int], str]:
    data = Path(path).read_bytes()
    if not data.startswith(b"!<arch>\n"):
        raise ValueError(f"not an ar archive: {path}")
    offset, occurrences, result = 8, {}, {}
    while offset < len(data):
        header = data[offset:offset + 60]
        if len(header) != 60 or header[58:60] != b"`\n":
            raise ValueError(f"malformed archive header at {offset}")
        raw_name = header[:16].decode("ascii").rstrip()
        size = int(header[48:58].decode("ascii").strip())
        payload = data[offset + 60:offset + 60 + size]
        if raw_name.startswith("#1/"):
            name_len = int(raw_name[3:])
            name, payload = payload[:name_len].decode("utf-8").rstrip("\0"), payload[name_len:]
        else:
            name = raw_name.rstrip("/")
        if not name.startswith("__.SYMDEF"):
            occurrence = occurrences.get(name, 0)
            occurrences[name] = occurrence + 1
            result[(name, occurrence)] = sha256_bytes(payload)
        offset += 60 + size + size % 2
    if offset != len(data):
        raise ValueError("archive scan did not end at file boundary")
    return result


def compile_command(name: str) -> list[str]:
    for line in COMPILE_LOG.read_text(errors="replace").splitlines():
        stripped = line.strip()
        if stripped.startswith("time mpif90 ") and stripped.endswith(f" {name}.f90"):
            argv = shlex.split(stripped)
            if argv[0] == "time":
                argv = argv[1:]
            return argv
    raise ValueError(f"saved compiler log has no command for {name}.f90")


def rewrite_compile(command: list[str], name: str, source: Path, output: Path, moddir: Path | None) -> list[str]:
    argv = list(command)
    for i, token in enumerate(argv):
        if token == "-o":
            argv[i + 1] = str(output)
            break
    else:
        raise ValueError(f"compile command for {name} has no output flag")
    argv[-1] = str(source)
    if moddir is not None:
        argv.extend(["-J", str(moddir), "-I", str(moddir)])
    return argv


def preprocess_authoritative_solve_f90(source_f: Path, destination_dir: Path) -> tuple[Path, dict]:
    """Replay the saved preprocessor pipeline from compile_cap3.log on pinned solve_em.F."""
    lines = COMPILE_LOG.read_text(errors="replace").splitlines()
    sed_line = next((x.strip() for x in lines if x.strip().startswith("sed -e")
                     and "solve_em.F > solve_em.G" in x), None)
    cpp_line = next((x.strip() for x in lines if x.strip().startswith("cpp ")
                     and "solve_em.G  > solve_em.bb" in x), None)
    standard_line = next((x.strip() for x in lines if "standard.exe solve_em.bb | cpp " in x
                          and "> solve_em.f90" in x), None)
    if not (sed_line and cpp_line and standard_line):
        raise ValueError("saved compile log lacks the complete solve_em preprocessing pipeline")
    destination_dir.mkdir(parents=True, exist_ok=True)
    g_file = destination_dir / "solve_em.G"
    bb_file = destination_dir / "solve_em.bb"
    f90_file = destination_dir / "solve_em.f90"

    sed_tokens = shlex.split(sed_line)
    redir = sed_tokens.index(">")
    sed_command = sed_tokens[:redir]
    sed_command[-1] = str(source_f)
    with g_file.open("wb") as stdout:
        result = subprocess.run(sed_command, cwd=SOURCE_DYN, stdout=stdout,
                                stderr=subprocess.PIPE, check=False)
    (destination_dir / "preprocess_sed.stderr.log").write_bytes(result.stderr)
    if result.returncode:
        raise subprocess.CalledProcessError(result.returncode, sed_command, result.stderr)

    cpp_tokens = shlex.split(cpp_line)
    cpp_redir = cpp_tokens.index(">")
    cpp_command = cpp_tokens[:cpp_redir]
    cpp_command[-1] = str(g_file)
    with bb_file.open("wb") as stdout:
        result = subprocess.run(cpp_command, cwd=SOURCE_DYN, stdout=stdout,
                                stderr=subprocess.PIPE, check=False)
    (destination_dir / "preprocess_cpp.stderr.log").write_bytes(result.stderr)
    if result.returncode:
        raise subprocess.CalledProcessError(result.returncode, cpp_command, result.stderr)

    pipeline_tokens = shlex.split(standard_line)
    pipe = pipeline_tokens.index("|")
    output_redir = pipeline_tokens.index(">")
    standard_command = pipeline_tokens[:pipe]
    cpp_after_standard = pipeline_tokens[pipe + 1:output_redir]
    standard_command[-1] = str(bb_file)
    if pipeline_tokens[output_redir + 1] != "solve_em.f90":
        raise ValueError("unexpected preprocessor output in saved log")
    standard_result = subprocess.run(standard_command, cwd=SOURCE_DYN,
                                     stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                                     check=False)
    if standard_result.returncode:
        raise subprocess.CalledProcessError(standard_result.returncode, standard_command,
                                            standard_result.stderr)
    with f90_file.open("wb") as stdout:
        cpp_result = subprocess.run(cpp_after_standard, cwd=SOURCE_DYN,
                                    input=standard_result.stdout, stdout=stdout,
                                    stderr=subprocess.PIPE, check=False)
    (destination_dir / "preprocess_standard.stderr.log").write_bytes(standard_result.stderr)
    (destination_dir / "preprocess_final_cpp.stderr.log").write_bytes(cpp_result.stderr)
    if cpp_result.returncode:
        raise subprocess.CalledProcessError(cpp_result.returncode, cpp_after_standard,
                                            cpp_result.stderr)
    commands = {
        "saved_log": str(COMPILE_LOG),
        "saved_log_sha256": sha256_file(COMPILE_LOG),
        "sed": sed_command,
        "cpp_pre_standard": cpp_command,
        "standard": standard_command,
        "cpp_post_standard": cpp_after_standard,
        "stderr_logs": [
            {"path": str(destination_dir / name), "sha256": sha256_file(destination_dir / name)}
            for name in ("preprocess_sed.stderr.log", "preprocess_cpp.stderr.log",
                         "preprocess_standard.stderr.log", "preprocess_final_cpp.stderr.log")
        ],
    }
    return f90_file, commands
