def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            digest.update(block)
    return digest.hexdigest()


def write_new(path: Path, value: dict) -> str:
    payload = (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + "\n").encode()
    with Path(path).open("xb") as stream:
        stream.write(payload)
        stream.flush()
        os.fsync(stream.fileno())
    return hashlib.sha256(payload).hexdigest()


def _pin(path: Path) -> dict:
    original = Path(path).expanduser()
    if original.is_symlink() or not original.is_file():
        raise ValueError(f"expected a pinned regular file: {original}")
    resolved = original.resolve(strict=True)
    return {"path": str(resolved), "sha256": sha256_file(resolved), "bytes": resolved.stat().st_size}

def load_and_verify_plan(plan_path: Path) -> tuple[dict, dict]:
    plan_path = Path(plan_path).resolve(strict=True)
    receipt_path = plan_path.with_name("plan_receipt.json")
    plan = json.loads(plan_path.read_text())
    receipt = json.loads(receipt_path.read_text())
    if receipt.get("plan_sha256") != sha256_file(plan_path):
        raise ValueError("plan receipt does not bind the frozen plan")
    if receipt.get("controller_sha256") != sha256_file(Path(__file__)):
        raise ValueError("run controller changed after plan preparation")
    if plan.get("schema") != "kdm6.native_moment_trace_v2_alpha025_conditional_pair_plan.v1":
        raise ValueError("unexpected conditional alpha0.25 plan schema")
    if plan.get("status") != "CONDITIONAL_PAIR_PREPARED_AWAITING_FOUR_CASE_RESULT_AND_ROOT_GO":
        raise ValueError("conditional plan status changed")
    for group in ("private_sources", "historical_inputs"):
        for label, item in plan[group].items():
            path = Path(item["path"])
            if not path.is_file() or path.is_symlink() or sha256_file(path) != item["sha256"]:
                raise ValueError(f"pinned {group} item changed: {label}")
    for asset in plan["runtime"]["assets"]:
        path = Path(asset["path"])
        if not path.is_file() or path.is_symlink() or path.stat().st_size != asset["bytes"]:
            raise ValueError(f"runtime asset missing/changed in size: {asset['case_basename']}")
        if sha256_file(path) != asset["sha256"]:
            raise ValueError(f"runtime asset hash changed: {asset['case_basename']}")
    for name, path_key, hash_key in (
        ("executable", "path", "sha256"), ("dso", "path", "sha256"),
    ):
        pin = plan["build"][name]
        path = Path(pin[path_key])
        if not path.is_file() or path.is_symlink() or sha256_file(path) != pin[hash_key]:
            raise ValueError(f"V2 build {name} changed after plan preparation")
    if plan["runtime"]["mode_flags"] != EXPECTED_ENV_MODES:
        raise ValueError("KDM mode selectors differ from the frozen plan")
    four_plan_receipt = json.loads(FOUR_CASE_RECEIPT.read_text())
    if (four_plan_receipt.get("plan_sha256") != plan["conditional_reason"]["four_case_plan_sha256"]
            or sha256_file(FOUR_CASE_PLAN) != plan["conditional_reason"]["four_case_plan_sha256"]):
        raise ValueError("first-four plan identity changed")
    return plan, receipt


def _checkpoint_state(checkpoint) -> object:
    import netCDF4
    import numpy as np

    dims = dict(checkpoint.schema[0])
    nx, ny, nz = (dims[name] for name in ("west_east", "south_north", "bottom_top"))
    with netCDF4.Dataset(checkpoint.file.path) as dataset:
        dataset.set_auto_maskandscale(False)
        fields = []
        for name in ("THM_2", "QVAPOR"):
            variable = dataset[name]
            if (variable.dimensions != ("Time", "bottom_top", "south_north", "west_east")
                    or np.dtype(variable.dtype) != np.dtype(np.float32)
                    or variable.shape != (1, nz, ny, nx)):
                raise ValueError(f"READY40 has invalid {name} layout")
            values = np.asarray(variable[0])
            if not np.isfinite(values).all():
                raise ValueError(f"READY40 {name} contains nonfinite values")
            fields.append(np.ascontiguousarray(values.transpose(1, 2, 0)).reshape(ny * nx, nz).astype(np.float64))
    return np.stack(fields)


def _ready_comparison(current, reference_path: Path, reference_sha: str) -> dict:
    import netCDF4
    from harness.native_host_transition import NativeCheckpoint
    from harness.native_replay_compare import compare_frames

    reference = NativeCheckpoint.capture(reference_path, expected_time="2025-07-19_00:00:40", expected_itimestep=2)
    if reference.file.sha256 != reference_sha:
        raise ValueError("historical control READY40 hash differs from plan")
    if current.checkpoint.schema != reference.schema:
        raise ValueError("new READY40 structure differs from historical control")
    with netCDF4.Dataset(reference_path) as baseline, netCDF4.Dataset(current.checkpoint.file.path) as candidate:
        baseline.set_auto_maskandscale(False)
        candidate.set_auto_maskandscale(False)
        comparison = compare_frames(baseline, candidate, 0, 0, "2025-07-19_00:00:40")
    comparison["new_ready_sha256"] = current.checkpoint.file.sha256
    comparison["reference_ready_sha256"] = reference_sha
    if comparison.get("raw_bitwise_equal") is not True:
        raise ValueError("new READY40 differs from the reference control checkpoint")
    return comparison


def verify_root_go(path: Path, *, plan_sha: str, run_id: str) -> dict:
    gate = json.loads(Path(path).read_text())
    if (gate.get("schema") != "kdm6.native_moment_trace_v2_execution_authorization.v1"
            or gate.get("status") != "APPROVED"
            or gate.get("plan_sha256") != plan_sha
            or run_id not in gate.get("approved_run_ids", [])
            or gate.get("four_case_result_status") != "COMPLETED_AND_REVIEWED"
            or gate.get("four_case_plan_sha256") != sha256_file(FOUR_CASE_PLAN)
            or gate.get("conditional_pair_reason_confirmed") is not True):
        raise RuntimeError("no root GO receipt is bound to this conditional pair and completed four-case review")
    return {"path": str(Path(path).resolve(strict=True)), "sha256": sha256_file(Path(path)),
            "status": gate["status"], "approved_run_id": run_id}


def execute_one(plan_path: Path, run_id: str, authorization_path: Path) -> dict:
    plan, plan_receipt = load_and_verify_plan(plan_path)
    case = next((row for row in plan["cases"] if row["run_id"] == run_id), None)
    if case is None:
        raise ValueError(f"run id is not in the plan: {run_id}")
    authorization = verify_root_go(authorization_path,
                                   plan_sha=plan_receipt["plan_sha256"], run_id=run_id)
    output = Path(case["output"])
    if output.exists() or output.is_symlink():
        raise FileExistsError(f"run output must be new: {output}")
    if not output.parent.is_dir():
        raise FileNotFoundError(f"run output parent is missing: {output.parent}")
    free = shutil.disk_usage(output.parent).free
    if free < 5 * (1 << 30):
        raise OSError(f"less than 5 GiB free before host run: {free} bytes")

    sys.path.insert(0, str(WORKTREE))
    from harness.native_host_transition import NativeCheckpoint, NativeInputs, PinnedFile
    from harness.native_live_stepper import NativeLiveStepper

    runtime = plan["runtime"]
    environment = dict(runtime["environment_without_v2_trace"])
    if V2_ENV in environment:
        raise ValueError("base environment already contains V2 trace variable")
    output.mkdir(parents=True, exist_ok=False)
    if case["v2_trace"] == "on":
        trace_directory = Path(case["trace_directory"])
        trace_directory.mkdir(parents=True, exist_ok=False)
        environment[V2_ENV] = str(trace_directory)
    else:
        trace_directory = None
        environment.pop(V2_ENV, None)

    initial = NativeCheckpoint.capture(
        runtime["origin"]["checkpoint_path"], expected_time=runtime["start_time"],
        expected_itimestep=runtime["origin"]["itimestep"],
    )
    if initial.file.sha256 != runtime["origin"]["checkpoint_sha256"]:
        raise ValueError("original 00:00:40 checkpoint changed during launch validation")
    external = NativeInputs(
        namelist=PinnedFile.bind(runtime["namelist_source_path"]),
        assets=tuple((row["case_basename"], PinnedFile.bind(row["path"]))
                     for row in runtime["assets"]),
        environment=tuple(sorted(environment.items())),
    )
    external.verify()
    executable = PinnedFile.bind(plan["build"]["executable"]["path"])
    library = PinnedFile.bind(plan["build"]["dso"]["path"])
    runner = PinnedFile.bind(WORKTREE / "harness/run_ss_case.py")

    write_new(output / "experiment_plan.json", plan)
    write_new(output / "launch_receipt.json", {
        "schema": "kdm6.native_moment_trace_v2_single_run_launch.v1",
        "status": "AUTHORIZED_SINGLE_HOST_STEP",
        "run_id": run_id,
        "plan_sha256": plan_receipt["plan_sha256"],
        "authorization": authorization,
        "trace_mode": case["v2_trace"],
        "execution_environment_sha256": hashlib.sha256(
            json.dumps(sorted(environment.items()), separators=(",", ":")).encode()
        ).hexdigest(),
        "free_disk_bytes_at_launch": free,
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "host_steps_requested": 1,
        "KDM6_wrapper_entries_expected": 2,
        "RTTOV_calls": 0,
    })

    stepper = None
    ready = None
    prepared = None
    error = None
    error_traceback = None
    try:
        stepper = NativeLiveStepper(
            initial=initial, external=external, executable=executable, library=library,
            runner=runner, work_root=output / "live_process", steps=1, dt_seconds=20,
            trace_column=tuple(runtime["trace_column_v1_i_j"]),
            radiation_every_step=True, control_family="ANALYSIS_TQ",
            capture_ready_checkpoint=True,
        )
        with stepper:
            current = stepper.current
            if current is None or current.ready is None:
                raise RuntimeError("live host did not expose an initial READY40")
            ready = _ready_comparison(
                current,
                Path(plan["historical_inputs"]["control_ready_checkpoint"]["path"]),
                plan["historical_inputs"]["control_ready_checkpoint"]["sha256"],
            )
            write_new(output / "ready_comparison.json", ready)
            if case["case"] == "control":
                result = stepper.advance(current, ())
                command_counts = {"applied_rows": 0, "applied_components": 0}
            else:
                import numpy as np
                from harness.native_analysis_increment import bind_satellite_control
                from harness.native_satellite_objective import prepare_representable_control

                with np.load(plan["historical_inputs"]["accepted_z"]["path"], allow_pickle=False) as archive:
                    original_z = np.asarray(archive["z"])
                z = original_z * float(plan["analysis"]["alpha"])
                if hashlib.sha256(z.tobytes()).hexdigest() != plan["analysis"]["requested_scaled_z_sha256"]:
                    raise ValueError("original accepted z was not scaled exactly once by alpha0.25")
                prepared = prepare_representable_control(
                    _checkpoint_state(current.checkpoint), z, z != 0.0,
                    horizontal_shape=(282, 234),
                    background_checkpoint_id=current.checkpoint.file.sha256,
                )
                with np.load(plan["historical_inputs"]["prepared_prediction"]["path"], allow_pickle=False) as archive:
                    if archive.files != ["candidate", "wire_delta", "effective_z"]:
                        raise ValueError("historical alpha0.25 prediction artifact schema differs")
                    predicted_candidate = np.asarray(archive["candidate"])
                    predicted_wire = np.asarray(archive["wire_delta"])
                    predicted_effective_z = np.asarray(archive["effective_z"])
                if (prepared["candidate"].tobytes() != predicted_candidate.tobytes()
                        or prepared["wire_delta"].tobytes() != predicted_wire.tobytes()
                        or prepared["z"].tobytes() != predicted_effective_z.tobytes()):
                    raise ValueError("READY40-derived alpha0.25 wire/effective direction differs from the prior pinned prediction")
                if prepared["counts"] != plan["analysis"]["prepare_counts"]:
                    raise ValueError("alpha0.25 preparation counts differ from the frozen prior prediction")
                if prepared["counts"].get("applied_points") != plan["analysis"]["expected_applied_points"]:
                    raise ValueError("alpha0.25 applied point count differs from the source-bound plan")
                if prepared["counts"].get("applied_components") != plan["analysis"]["expected_applied_components"]:
                    raise ValueError("alpha0.25 applied component count differs from the source-bound plan")
                if hashlib.sha256(prepared["z"].tobytes()).hexdigest() != plan["analysis"]["effective_z_sha256"]:
                    raise ValueError("alpha0.25 effective z differs from the frozen representable prediction")
                increment = bind_satellite_control(
                    current.checkpoint, prepared,
                    posterior_checkpoint_id="historical PR448 microphysics-only proposal; alpha=0.25; no full-host acceptance",
                )
                command_counts = {
                    "applied_rows": int(prepared["counts"]["applied_points"]),
                    "applied_components": int(prepared["counts"]["applied_components"]),
                    "wire_delta_sha256": hashlib.sha256(prepared["wire_delta"].tobytes()).hexdigest(),
                    "candidate_sha256": hashlib.sha256(prepared["candidate"].tobytes()).hexdigest(),
                }
                result = stepper.advance(current, increment)
            if result.checkpoint.time != runtime["end_time"] or result.checkpoint.itimestep != 3:
                raise RuntimeError("host run ended at an unexpected time or timestep")
    except BaseException as exc:
        error = exc
        error_traceback = traceback.format_exc()

    stepper_record = None if stepper is None else stepper._record
    trace_files = {}
    if trace_directory is not None:
        for name in ("host.hex", "kernel.hex"):
            path = trace_directory / name
            trace_files[name] = ({"path": str(path), "sha256": sha256_file(path), "bytes": path.stat().st_size}
                                 if path.is_file() else None)
    record = {
        "schema": "kdm6.native_moment_trace_v2_single_run_result.v1",
        "status": "FAILED" if error is not None else "STEP_COMPLETE",
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "run_id": run_id,
        "plan_sha256": plan_receipt["plan_sha256"],
        "controller_sha256": sha256_file(Path(__file__)),
        "error": None if error is None else f"{type(error).__name__}: {error}",
        "traceback": error_traceback,
        "ready_comparison": ready,
        "command_counts": command_counts if error is None else None,
        "live_session_record": stepper_record,
        "native_host_steps_completed": 0 if stepper_record is None else sum(
            item.get("output_time") is not None for item in stepper_record.get("controls", [])
        ),
        "analysis_wire_sha256": None if not stepper_record or not stepper_record.get("controls") else
            stepper_record["controls"][-1].get("command_sha256"),
        "v2_trace_directory": None if trace_directory is None else str(trace_directory),
        "v2_trace_files": trace_files,
        "v1_trace": None if stepper_record is None else stepper_record.get("native_trace"),
        "output_time": None if stepper_record is None or not stepper_record.get("controls") else
            stepper_record["controls"][-1].get("output_time"),
        "output_checkpoint_sha256": None if stepper_record is None or not stepper_record.get("controls") else
            stepper_record["controls"][-1].get("output_checkpoint_sha256"),
        "model_calls": {"host_steps_expected": 1, "KDM6_wrapper_entries_expected": 2,
                        "RTTOV": 0, "optimizer": 0},
    }
    write_new(output / "run_result.json", record)
    if error is not None:
        raise error
    return record


def main(argv: list[str] | None = None) -> int:
    import argparse

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--prepare", action="store_true", help="freeze a conditional two-run alpha0.25 pair plan")
    parser.add_argument("--plan", type=Path, default=PLAN_PATH)
    parser.add_argument("--run-id", choices=RUN_IDS)
    parser.add_argument("--authorization", type=Path, default=AUTHORIZATION_PATH)
    parser.add_argument("--execute", action="store_true", help="run exactly one case; requires root GO receipt")
    args = parser.parse_args(argv)
    if args.prepare:
        print(json.dumps(prepare_plan(), sort_keys=True))
        return 0
    if not args.run_id:
        parser.error("--run-id is required when --prepare is not set")
    if not args.execute:
        _, receipt = load_and_verify_plan(args.plan)
        print(json.dumps({"status": "PREFLIGHT_PASS_NOT_EXECUTED", "run_id": args.run_id,
                          "plan_sha256": receipt["plan_sha256"], "host_steps": 0,
                          "KDM6_calls": 0, "RTTOV_calls": 0}, sort_keys=True))
        return 0
    result = execute_one(args.plan, args.run_id, args.authorization)
    print(json.dumps({"status": result["status"], "run_id": args.run_id,
                      "host_steps": result["native_host_steps_completed"],
                      "output_checkpoint_sha256": result["output_checkpoint_sha256"]}, sort_keys=True))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
