"""Private fixed-M0 direct RTTOV diagnostic; no host, KDM, K, or optimizer calls.

Requires completed matched pilot-on trace-off/on runs. Explicit --execute is
required. Each chunk has a fresh, immutable attempt directory; a partial chunk
is never silently retried. Resume validates every completed chunk and its pins.
Costs are undefined unless every original M0 cell remains valid.
After --execute, binding.json is written before support preflight; that stage
is not filesystem read-only. Completed receipt counts do not include failed
attempts which may have called RTTOV. Raw number units remain uncertified.
"""

def sha(p):
 h=hashlib.sha256()
 with Path(p).open('rb') as f:
  for b in iter(lambda:f.read(8<<20),b''): h.update(b)
 return h.hexdigest()

def source_tree_pin():
 """Pin operational harness/oracle code and bundled observation data."""
 files=sorted([*REPO.glob('harness/*.py'),*REPO.glob('oracle/**/*.py'),
               *REPO.glob('oracle/kdm6/obs/data/**/*')])
 files=[p for p in files if p.is_file() and '__pycache__' not in p.parts]
 h=hashlib.sha256()
 for p in files:
  h.update(p.relative_to(REPO).as_posix().encode()+b'\0'+sha(p).encode()+b'\n')
 return len(files),h.hexdigest()

def new_json(p, x):
 with Path(p).open('x') as f:
  json.dump(x,f,sort_keys=True,indent=2,allow_nan=False); f.write('\n'); f.flush(); os.fsync(f.fileno())
