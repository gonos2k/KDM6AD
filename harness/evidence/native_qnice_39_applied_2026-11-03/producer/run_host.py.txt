def digest(path: Path | str) -> str:
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(8 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def verify_cpp_build(receipt_path: Path, library: Path, worktree: Path) -> dict:
    receipt = json.loads(receipt_path.read_bytes())
    if receipt.get("schema") != "kdm6.native_qnice_39_cpp_build.v1":
        raise ValueError("unexpected C++ build receipt schema")
    commit = subprocess.check_output(
        ["git", "rev-parse", "HEAD"], cwd=worktree, text=True).strip()
    if commit != receipt["public_worktree_commit"]:
        raise ValueError("public checkout differs from C++ build receipt")
    expected_dso = receipt["dso_sha256"]
    for path in (library, Path(receipt["dso_path"]), Path(receipt["staged_dso_path"])):
        if digest(path) != expected_dso:
            raise ValueError("C++ build DSO identity differs: " + str(path))
    if (receipt["staged_dso_sha256"] != expected_dso or
            receipt["staged_dso_byte_identical_to_build"] is not True or
            digest(ROOT / "cpp_build/build/libkdm6.a") != receipt["archive_sha256"] or
            digest(receipt["parent_c5_receipt_path"]) != receipt["parent_c5_receipt_sha256"]):
        raise ValueError("C++ archive/staging/parent receipt differs")
    source_root = ROOT / "cpp_build/source"
    for relative, expected in receipt["instrumented_source_sha256"].items():
        if digest(source_root / relative) != expected:
            raise ValueError("instrumented C++ source differs: " + relative)
    for relative, expected in receipt["current_public_source_sha256"].items():
        if digest(worktree / "libtorch" / relative) != expected:
            raise ValueError("public C++ source differs: " + relative)
    object_dir = ROOT / "cpp_build/build/CMakeFiles"
    for name, expected in receipt["instrumented_object_sha256"].items():
        kind = "kdm6_c.dir/bridge" if name == "kdm6_c_api.cpp.o" else "kdm6.dir/src"
        if digest(object_dir / kind / name) != expected:
            raise ValueError("instrumented C++ object differs: " + name)
    if (receipt["verification"]["host_run_performed"] is not False or
            receipt["verification"]["all_required_ice_fields_cover_39_K"] is not True or
            receipt["verification"]["historical_c5_scalar_stream_byte_identical"] is not True):
        raise ValueError("C++ build preflight did not pass")
    return receipt

def verify_host_build(receipt_path: Path, executable: Path, library: Path) -> dict:
    receipt = json.loads(receipt_path.read_bytes())
    if receipt.get("schema") != "kdm6ad.native_qnice_host_build.v1" or \
            receipt.get("host_case_run_performed") is not False:
        raise ValueError("unexpected host build receipt schema/status")
    for section, key, expected_path in (
        ("executable", "sha256", executable),
        ("cxx_dso", "sha256", library),
    ):
        record = receipt[section]
        if Path(record["path"]).resolve() != expected_path.resolve() or \
                digest(expected_path) != record[key]:
            raise ValueError("host build binary binding differs: " + section)
    if (receipt["cxx_dso"]["sha256"] != receipt["cxx_dso"]["expected_sha256"] or
            receipt["cxx_dso"]["sha256"] != receipt["cxx_dso"]["cpp_build_receipt"]["staged_dso_sha256"] or
            digest(receipt["cxx_dso"]["cpp_build_receipt"]["path"]) != CPP_RECEIPT_SHA or
            Path(receipt["executable"]["expected_runtime_dso_realpath"]).resolve() != library.resolve() or
            receipt["executable"]["linked_dso_install_name"] != "@rpath/libkdm6_c.2.dylib"):
        raise ValueError("host/C++ DSO lineage differs")
    for record in (receipt["source_copy"]["canonical_source"],
                   receipt["source_copy"]["instrumented_source"],
                   receipt["source_copy"]["copied_build_prerequisite"],
                   receipt["fortran"]["solve_em_object"],
                   receipt["archive"],
                   receipt["link"]["link_log"],
                   receipt["link"]["link_exit_file"]):
        if digest(record["path"]) != record["sha256"]:
            raise ValueError("host build source/object/archive/log pin differs: " + record["path"])
    archive = receipt["archive"]
    if (digest(archive["canonical_archive_path"]) != archive["canonical_archive_sha256"] or
            archive["member_count"] != 376 or archive["canonical_member_count"] != 376 or
            archive["member_inventory_equal"] is not True or
            archive["only_replaced_member"] != "solve_em.o" or

            receipt["fortran"]["kdm6_fortran_fp_contract_rule"] != "-ffp-contract=off"):
        raise ValueError("host build archive/flags contract differs")
    return receipt

def main() -> None:

    parser.add_argument("--worktree", type=Path, required=True)

    parser.add_argument("--library", type=Path, required=True)
    parser.add_argument("--cpp-receipt", type=Path, required=True)
    parser.add_argument("--host-receipt", type=Path, required=True)
    parser.add_argument("--mode", choices=("off", "on", "off_fp64", "on_fp64",
                                          "off_conservative", "on_conservative",
                                          "off_final", "on_final"), required=True)
    parser.add_argument("--batch-index", type=int, default=22617)
    args = parser.parse_args()

    if digest(PLAN) != PLAN_SHA:
        raise ValueError("historical input plan differs")
    runtime = json.loads(PLAN.read_bytes())["runtime"]
    inputs = [(runtime["origin"]["checkpoint_path"], runtime["origin"]["checkpoint_sha256"]),
              (runtime["namelist_source_path"], runtime["namelist_source_sha256"])]
    inputs += [(item["path"], item["sha256"]) for item in runtime["assets"]]
    for path, expected in inputs:
        if digest(path) != expected:
            raise ValueError("host input pin differs: " + path)
    for name in ("OMP_NUM_THREADS", "MKL_NUM_THREADS", "OMP_THREAD_LIMIT", "VECLIB_MAXIMUM_THREADS"):
        if runtime["environment_without_trace"].get(name) != "1":
            raise ValueError("one-rank one-thread environment differs: " + name)
    if shutil.disk_usage(ROOT).free < 8 * (1 << 30):
        raise OSError("less than 8 GiB free for full-domain host step")
    for path in (args.exe, args.library, args.cpp_receipt, args.host_receipt):
        if not path.is_file():
            raise FileNotFoundError(path)
    if args.mode.endswith("final"):
        if digest(args.cpp_receipt) != CPP_ATTEMPT4_RECEIPT_SHA:
            raise ValueError("identity-fix C++ receipt pin differs")
        cpp_receipt = verify_cpp_attempt4(args.cpp_receipt, args.library)
        host_receipt = verify_host_attempt5(args.host_receipt, args.exe, args.library)
    elif args.mode.endswith("conservative"):
        if digest(args.cpp_receipt) != CPP_ATTEMPT2_RECEIPT_SHA:
            raise ValueError("conservative C++ build receipt pin differs")
        cpp_receipt = verify_cpp_attempt2(args.cpp_receipt, args.library)
        if digest(args.host_receipt) != HOST_ATTEMPT3_RECEIPT_SHA:
            raise ValueError("attempt3 host receipt pin differs")
        host_receipt = verify_host_attempt3(args.host_receipt, args.exe, args.library)
    elif args.mode.endswith("fp64"):
        if digest(args.cpp_receipt) != CPP_RECEIPT_SHA:
            raise ValueError("C++ build receipt pin differs")
        cpp_receipt = verify_cpp_build(args.cpp_receipt, args.library, args.worktree.resolve())
        if digest(args.host_receipt) != HOST_RETRY_RECEIPT_SHA:
            raise ValueError("historical host retry receipt pin differs")
        host_receipt = verify_host_retry(args.host_receipt, args.exe, args.library)
    else:
        if digest(args.cpp_receipt) != CPP_RECEIPT_SHA:
            raise ValueError("C++ build receipt pin differs")
        cpp_receipt = verify_cpp_build(args.cpp_receipt, args.library, args.worktree.resolve())
        if digest(args.host_receipt) != HOST_RECEIPT_SHA:
            raise ValueError("host build receipt pin differs")
        host_receipt = verify_host_build(args.host_receipt, args.exe, args.library)

    run = ROOT / "runs" / args.mode
    run.mkdir(parents=True, exist_ok=False)
    scratch = run / "scratch"
    scratch.mkdir()
    capture = run / "capture"
    if args.mode.startswith("on"):
        capture.mkdir()

    sys.path.insert(0, str(args.worktree.resolve()))
    from harness.native_host_transition import NativeCheckpoint, NativeInputs, PinnedFile
    from harness.native_live_stepper import NativeLiveStepper

    origin = NativeCheckpoint.capture(
        runtime["origin"]["checkpoint_path"], expected_time=runtime["start_time"],
        expected_itimestep=runtime["origin"]["itimestep"])
    if origin.file.sha256 != runtime["origin"]["checkpoint_sha256"]:
        raise ValueError("origin checkpoint identity differs")
    env = dict(runtime["environment_without_trace"])
    env["KDM6AD_QNC_LIVE_GUARD"] = "1"
    if args.mode.startswith("on"):
        env.update(
            KDM6_NATIVE_CAPTURE_DIR=str(capture.resolve()),
            KDM6_NATIVE_CAPTURE_CALL="2",
            KDM6_NATIVE_CAPTURE_B=str(args.batch_index),
            KDM6_NATIVE_CAPTURE_ALL_K="1",
            KDM6_NATIVE_CAPTURE_LEGACY_SCALAR="1",
            KDM6_NATIVE_CAPTURE_K1="20",
            KDM6_NATIVE_CAPTURE_K2="22",
            KDM6_NATIVE_CAPTURE_MAX_ROWS="5000",
            KDM6_NATIVE_CAPTURE_MAX_BYTES=str(1 << 20),
            KDM6_NATIVE_CAPTURE_LEGACY_MAX_ROWS="450000",
            KDM6_NATIVE_CAPTURE_LEGACY_MAX_BYTES=str(64 << 20),
            KDM6AD_NATIVE_QNICE_TRACE=str((capture / "host_qnice_pre_call.csv").resolve()),
        )
    external = NativeInputs(
        namelist=PinnedFile.bind(runtime["namelist_source_path"]),
        assets=tuple((item["case_basename"], PinnedFile.bind(item["path"]))
                     for item in runtime["assets"]),
        environment=tuple(sorted(env.items())),
    )
    with NativeLiveStepper(
        initial=origin, external=external,
        executable=PinnedFile.bind(args.exe.resolve()),
        library=PinnedFile.bind(args.library.resolve()),
        runner=PinnedFile.bind(args.worktree.resolve() / "harness/run_ss_case.py"),
        work_root=scratch, steps=1, dt_seconds=20,
        trace_column=(164, 242), radiation_every_step=True,
        control_family="ANALYSIS_TQ", capture_ready_checkpoint=True,
    ) as stepper:
        ready = stepper.current
        if ready is None or ready.ready is None:

        final = stepper.advance(ready, ())
        result = {
