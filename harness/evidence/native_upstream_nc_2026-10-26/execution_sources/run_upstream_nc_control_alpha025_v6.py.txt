def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 << 20), b""):
            digest.update(block)
    return digest.hexdigest()


def write_new(path: Path, value: dict) -> str:
    payload = (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + "\n").encode()
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("xb") as stream:
        stream.write(payload)
        stream.flush()
        os.fsync(stream.fileno())
    return hashlib.sha256(payload).hexdigest()


def pin(path: Path) -> dict:
    path = Path(path).expanduser()
    if path.is_symlink() or not path.is_file():
        raise ValueError(f"expected a pinned regular file: {path}")
    resolved = path.resolve(strict=True)
    return {"path": str(resolved), "sha256": sha256_file(resolved),
            "bytes": resolved.stat().st_size}

    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


def _one_live_directory(run_dir: Path) -> Path:
    matches = sorted((run_dir / "live_process").glob("live_*"))
    if len(matches) != 1 or not matches[0].is_dir():
        raise ValueError(f"expected one closed live-process directory under {run_dir}")
    return matches[0]


def _final_checkpoint_path(run_dir: Path, expected_sha: str) -> Path:
    matches = sorted((run_dir / "live_process").glob("live_*/wrfrst_d01_2025-07-19_00:01:00"))
    if len(matches) != 1 or not matches[0].is_file():
        raise ValueError(f"missing unique completed checkpoint in {run_dir}")
    if sha256_file(matches[0]) != expected_sha:
        raise ValueError(f"historical checkpoint SHA differs from run result in {run_dir}")
    return matches[0].resolve(strict=True)


def _ack_path(run_dir: Path, expected_sha: str) -> Path:
    live = _one_live_directory(run_dir)
    path = live / "control/applied_0000000002.txt"
    if not path.is_file() or sha256_file(path) != expected_sha:
        raise ValueError(f"historical ACK file does not match the completed run in {run_dir}")
    return path.resolve(strict=True)


def _result(run_dir: Path) -> dict:
    result = json.loads((run_dir / "run_result.json").read_text())
    if result.get("status") != "STEP_COMPLETE" or result.get("native_host_steps_completed") != 1:
        raise ValueError(f"historical native run is incomplete: {run_dir}")
    return result


def _v4_control_off_result() -> dict:
    result_path = V4_CONTROL_OFF_DIR / "run_result.json"
    result = json.loads(result_path.read_text())
    if (result.get("status") != "RUN_COMPLETE_FULL_COMPARISONS_PASS"
            or result.get("trace_mode") != "off"
            or result.get("run_id") != "control_upstream_nc_v4_off"
            or result.get("native_host_steps_completed") != 1):
        raise ValueError("external V4 control-off result is not a completed single host step")
    for path_key, sha_key, flag in (
        ("ready_comparison_path", "ready_comparison_sha256", "raw_bitwise_equal"),
        ("endpoint_comparison_path", "endpoint_comparison_sha256", "raw_bitwise_equal"),
        ("ack_comparison_path", "ack_comparison_sha256", "byte_identical_to_prior_same_case"),
    ):
        path = Path(result[path_key])
        if not path.is_file() or sha256_file(path) != result[sha_key]:
            raise ValueError(f"external V4 control-off {path_key} artifact changed")
        comparison = json.loads(path.read_text())
        if comparison.get(flag) is not True:
            raise ValueError(f"external V4 control-off {path_key} did not pass")
    live_path = Path(result["live_session_record_path"])
    if not live_path.is_file() or sha256_file(live_path) != result["live_session_record_sha256"]:
        raise ValueError("external V4 control-off live-session record changed")
    live = json.loads(live_path.read_text())
    if (not live.get("controls")
            or live["controls"][-1].get("ack_sha256") != result.get("ack_sha256")
            or live["controls"][-1].get("output_checkpoint_sha256") != result.get("output_checkpoint_sha256")):
        raise ValueError("external V4 control-off result differs from its session record")
    run_plan = json.loads((V4_CONTROL_OFF_DIR / "experiment_plan.json").read_text())
    if result.get("plan_sha256") != sha256_file(V4_CONTROL_OFF_DIR / "experiment_plan.json"):
        raise ValueError("external V4 control-off result does not match its saved experiment plan")
    if run_plan["build"]["receipt"]["sha256"] != sha256_file(V4_BUILD_RECEIPT):
        raise ValueError("external V4 control-off run used a different private build")
    return result

def _load_plan(path: Path) -> tuple[dict, dict]:
    path = Path(path).resolve(strict=True)
    receipt_path = path.with_name("plan_receipt.json")
    plan = json.loads(path.read_text())
    receipt = json.loads(receipt_path.read_text())
    if receipt.get("plan_sha256") != sha256_file(path):
        raise ValueError("V4 plan receipt does not bind the plan")
    if receipt.get("controller_sha256") != sha256_file(Path(__file__)):
        raise ValueError("V4 controller changed after plan preparation")
    if plan.get("schema") != "kdm6.native_upstream_nc_trace_v6_remaining_control_alpha025_plan.v1":
        raise ValueError("unexpected V6 remaining control/alpha plan schema")
    for group in ("private_sources", "historical_inputs"):
        for label, item in plan[group].items():
            source = Path(item["path"])
            if not source.is_file() or source.is_symlink() or sha256_file(source) != item["sha256"]:
                raise ValueError(f"V4 plan source pin changed: {group}/{label}")
    for key in ("executable", "dso"):
        item = plan["build"][key]
        source = Path(item["path"])
        if not source.is_file() or source.is_symlink() or sha256_file(source) != item["sha256"]:
            raise ValueError(f"V4 private {key} changed after plan preparation")
    return plan, receipt


def _verify_go(path: Path, plan_sha: str, run_id: str) -> dict:
    gate = json.loads(path.read_text())
    if (gate.get("schema") != "kdm6.native_upstream_nc_trace_v6_execution_authorization.v1"
            or gate.get("status") != "APPROVED"
            or gate.get("plan_sha256") != plan_sha
            or run_id not in gate.get("approved_run_ids", [])):
        raise RuntimeError("no GO receipt is bound to this V4 plan and run id")
    return {"path": str(path.resolve(strict=True)), "sha256": sha256_file(path),
            "status": gate["status"], "approved_run_id": run_id}


def _compare_full(reference_path: Path, reference_sha: str, current_checkpoint) -> dict:
    import netCDF4
    from harness.native_host_transition import NativeCheckpoint
    from harness.native_replay_compare import compare_frames
    reference = NativeCheckpoint.capture(reference_path, expected_time="2025-07-19_00:01:00", expected_itimestep=3)
    if reference.file.sha256 != reference_sha:
        raise ValueError("prior final checkpoint hash differs from the pinned reference")
    if current_checkpoint.schema != reference.schema:
        raise ValueError("V4 endpoint structure differs from the prior same-case endpoint")
    with netCDF4.Dataset(reference_path) as baseline, netCDF4.Dataset(current_checkpoint.file.path) as candidate:
        comparison = compare_frames(baseline, candidate, 0, 0, "2025-07-19_00:01:00")
    comparison.update({"new_endpoint_sha256": current_checkpoint.file.sha256,
                       "reference_endpoint_sha256": reference_sha,
                       "variable_count": len(reference.schema[1])})
    if comparison.get("raw_bitwise_equal") is not True:
        raise ValueError("V4 endpoint differs from the prior same-case full checkpoint")
    return comparison


def execute_one(plan_path: Path, run_id: str, authorization_path: Path) -> dict:
    plan, plan_receipt = _load_plan(plan_path)
    case = next((item for item in plan["cases"] if item["run_id"] == run_id), None)
    if case is None:
        raise ValueError(f"run id is not in the frozen plan: {run_id}")
    authorization = _verify_go(authorization_path, plan_receipt["plan_sha256"], run_id)
    output = Path(case["output"])
    if output.exists() or output.is_symlink():
        raise FileExistsError(f"V4 run output must be new: {output}")
    if not output.parent.is_dir():
        raise FileNotFoundError(output.parent)
    free = shutil.disk_usage(output.parent).free
    if free < 5 * (1 << 30):
        raise OSError(f"less than 5 GiB free before V4 run: {free} bytes")

    sys.path.insert(0, str(WORKTREE))
    from harness.native_host_transition import NativeCheckpoint, NativeInputs, PinnedFile
    from harness.native_live_stepper import NativeLiveStepper

    runtime = plan["runtime"]
    environment = dict(runtime["environment_without_trace"])
    upstream_env = plan["upstream_trace"]["environment_variable"]
    if V2_ENV in environment or upstream_env in environment:
        raise ValueError("base environment already contains a trace output directory")
    v2_dir = None if case["v2_trace_directory"] is None else Path(case["v2_trace_directory"])
    upstream_dir = None if case["upstream_trace_directory"] is None else Path(case["upstream_trace_directory"])
    if case["trace_mode"] == "on":
        if v2_dir.exists() or upstream_dir.exists():
            raise FileExistsError("V4 trace directories must be new")
        environment[V2_ENV] = str(v2_dir)
        environment[upstream_env] = str(upstream_dir)
    else:
        environment.pop(V2_ENV, None)
        environment.pop(upstream_env, None)

    initial = NativeCheckpoint.capture(runtime["origin"]["checkpoint_path"],
                                       expected_time=runtime["start_time"],
                                       expected_itimestep=runtime["origin"]["itimestep"])
    if initial.file.sha256 != runtime["origin"]["checkpoint_sha256"]:
        raise ValueError("original restart changed before V4 launch")
    external = NativeInputs(
        namelist=PinnedFile.bind(runtime["namelist_source_path"]),
        assets=tuple((row["case_basename"], PinnedFile.bind(row["path"])) for row in runtime["assets"]),
        environment=tuple(sorted(environment.items())),
    )
    external.verify()
    executable = PinnedFile.bind(plan["build"]["executable"]["path"])
    library = PinnedFile.bind(plan["build"]["dso"]["path"])
    runner = PinnedFile.bind(WORKTREE / "harness/run_ss_case.py")

    output.mkdir(parents=True, exist_ok=False)
    if case["trace_mode"] == "on":
        output_root = output.resolve(strict=True)
        for directory in (v2_dir, upstream_dir):
            target = directory.resolve(strict=False)
            if not target.is_relative_to(output_root):
                raise ValueError("V4 trace directories must be children of their new run output")
            directory.mkdir(parents=True, exist_ok=False)
    write_new(output / "experiment_plan.json", plan)
    write_new(output / "launch_receipt.json", {
        "schema": "kdm6.native_upstream_nc_v6_single_run_launch.v1",
        "status": "AUTHORIZED_SINGLE_HOST_STEP",
        "run_id": run_id,
        "plan_sha256": plan_receipt["plan_sha256"],
        "authorization": authorization,
        "trace_mode": case["trace_mode"],
        "free_disk_bytes_at_launch": free,
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "host_steps_requested": 1,
        "KDM6_wrapper_entries_expected": 2,
        "RTTOV_calls": 0,
    })

    stepper = None
    error = None
    error_traceback = None
    ready_comparison = None
    endpoint_comparison = None
    command_summary = None
    endpoint_path = None
    expected_ack = Path(plan["historical_inputs"][case["case"] + "_ack_reference"]["path"])
    reference_final = plan["historical_inputs"][case["case"] + "_final_reference"]
    try:
        stepper = NativeLiveStepper(
            initial=initial, external=external, executable=executable, library=library,
            runner=runner, work_root=output / "live_process", steps=1, dt_seconds=20,
            trace_column=tuple(runtime["trace_column_v1_i_j"]),
            radiation_every_step=True, control_family="ANALYSIS_TQ",
            capture_ready_checkpoint=True,
        )
        with stepper:
            current = stepper.current
            if current is None or current.ready is None:
                raise RuntimeError("live V4 host did not expose READY40")
            legacy = load_legacy_helpers()
            ready_comparison = legacy._ready_comparison(
                current,
                Path(plan["historical_inputs"]["ready40_reference"]["path"]),
                plan["historical_inputs"]["ready40_reference"]["sha256"],
            )
            write_new(output / "ready_comparison.json", ready_comparison)
            if case["case"] == "control":
                result_state = stepper.advance(current, ())
                command_summary = {"applied_rows": 0, "applied_components": 0}
            else:
                import numpy as np
                from harness.native_analysis_increment import bind_satellite_control
                from harness.native_satellite_objective import prepare_representable_control
                alpha_plan = json.loads(Path(plan["historical_inputs"]["alpha025_plan"]["path"]).read_text())
                with np.load(plan["historical_inputs"]["accepted_z"]["path"], allow_pickle=False) as archive:
                    original_z = np.asarray(archive["z"])
                scaled_z = original_z * float(plan["analysis"]["alpha"])
                if hashlib.sha256(scaled_z.tobytes()).hexdigest() != plan["analysis"]["requested_scaled_z_sha256"]:
                    raise ValueError("alpha0.25 direction differs from the pinned source wire")
                prepared = prepare_representable_control(
                    legacy._checkpoint_state(current.checkpoint), scaled_z, scaled_z != 0.0,
                    horizontal_shape=(282, 234),
                    background_checkpoint_id=current.checkpoint.file.sha256,
                )
                with np.load(plan["historical_inputs"]["prepared_prediction"]["path"], allow_pickle=False) as archive:
                    predicted = {name: np.asarray(archive[name]) for name in ("candidate", "wire_delta", "effective_z")}
                if (prepared["candidate"].tobytes() != predicted["candidate"].tobytes()
                        or prepared["wire_delta"].tobytes() != predicted["wire_delta"].tobytes()
                        or prepared["z"].tobytes() != predicted["effective_z"].tobytes()):
                    raise ValueError("READY40 alpha0.25 wire differs from the source-bound prediction")
                if prepared["counts"] != plan["analysis"]["prepare_counts"]:
                    raise ValueError("alpha0.25 preparation counts changed")
                if prepared["counts"].get("applied_points") != plan["analysis"]["expected_applied_points"]:
                    raise ValueError("alpha0.25 applied point count changed")
                if prepared["counts"].get("applied_components") != plan["analysis"]["expected_applied_components"]:
                    raise ValueError("alpha0.25 applied component count changed")
                increment = bind_satellite_control(
                    current.checkpoint, prepared,
                    posterior_checkpoint_id="historical alpha0.25 diagnostic proposal; not an accepted update",
                )
                command_summary = {
                    "applied_rows": int(prepared["counts"]["applied_points"]),
                    "applied_components": int(prepared["counts"]["applied_components"]),
                    "candidate_sha256": hashlib.sha256(prepared["candidate"].tobytes()).hexdigest(),
                    "wire_delta_sha256": hashlib.sha256(prepared["wire_delta"].tobytes()).hexdigest(),
                }
                result_state = stepper.advance(current, increment)
            if result_state.checkpoint.time != runtime["end_time"] or result_state.checkpoint.itimestep != 3:
                raise RuntimeError("V4 host ended at an unexpected time or timestep")
            endpoint_path = Path(result_state.checkpoint.file.path).resolve(strict=True)
            endpoint_comparison = _compare_full(Path(reference_final["path"]),
                                                reference_final["sha256"],
                                                result_state.checkpoint)
            write_new(output / "endpoint_comparison.json", endpoint_comparison)
    except BaseException as exc:
        error = exc
        error_traceback = traceback.format_exc()

    live_record = None if stepper is None else stepper._record
    live_record_path = output / "live_session_record.json"
    if live_record is not None:
        write_new(live_record_path, live_record)
    ack_path = None if stepper is None else stepper.control_root / "applied_0000000002.txt"
    ack_report = None
    if ack_path is not None and ack_path.is_file():
        current_ack = ack_path.read_bytes()
        expected_ack_bytes = expected_ack.read_bytes()
        ack_report = {
            "path": str(ack_path), "sha256": sha256_file(ack_path), "bytes": len(current_ack),
            "reference_path": str(expected_ack),
            "reference_sha256": sha256_file(expected_ack),
            "byte_identical_to_prior_same_case": current_ack == expected_ack_bytes,
        }
        write_new(output / "ack_comparison.json", ack_report)

    upstream_path = None if upstream_dir is None else upstream_dir / "upstream_nc.tsv"
    upstream_rows = []
    if upstream_path is not None and upstream_path.is_file():
        lines = upstream_path.read_text().splitlines()
        if (len(lines) < 2 or lines[0] != UPSTREAM_SCHEMA
                or tuple(lines[1].lstrip("# ").split()) != UPSTREAM_COLUMNS):
            raise ValueError("upstream NC trace header differs from the pinned V1 schema")
        upstream_rows = [line for line in lines[2:] if line and not line.startswith("#")]
        for row in upstream_rows:
            fields = row.split()
            if len(fields) != 10 or fields[0] != "VAL":
                raise ValueError("malformed upstream NC trace row")
            if fields[7] == "TENDEC":
                if fields[8] != "bool" or fields[9] not in {"00", "01"}:
                    raise ValueError("malformed TENDEC trace value")
            elif fields[8] != "f32" or len(fields[9]) != 8:
                raise ValueError("malformed REAL4 upstream NC trace value")
    trace_files = {}
    if case["trace_mode"] == "on":
        for label, path in (("host.hex", v2_dir / "host.hex"),
                            ("kernel.hex", v2_dir / "kernel.hex"),
                            ("upstream_nc.tsv", upstream_path)):
            if path.is_file():
                trace_files[label] = {"path": str(path), "bytes": path.stat().st_size, "sha256": sha256_file(path)}
    if error is None:
        try:
            if case["trace_mode"] == "on":
                if set(trace_files) != {"host.hex", "kernel.hex", "upstream_nc.tsv"}:
                    raise RuntimeError("V4 trace files are incomplete")
                if not upstream_rows:
                    raise RuntimeError("V4 upstream NC trace produced no selected rows")
                stage_targets = {
                    (row.split()[3], (int(row.split()[4]), int(row.split()[6]), int(row.split()[5])))
                    for row in upstream_rows
                }
                required_stage_targets = {
                    (stage, target)
                    for stage in ("RKSCALAR_BEFORE", "RKSCALAR_AFTER", "RKSCALAR_PD_BEFORE",
                                  "RKSCALAR_PD_AFTER", "PRE_MICROPHYSICS")
                    for target in TARGETS
                }
                if not required_stage_targets.issubset(stage_targets):
                    missing = sorted(required_stage_targets - stage_targets)
                    raise RuntimeError(f"V4 upstream trace omitted expected stage/target rows: {missing}")
            elif trace_files or V2_ENV in environment or upstream_env in environment:
                raise RuntimeError("V4 trace-off environment unexpectedly emitted trace files")
            if not ack_report or not ack_report["byte_identical_to_prior_same_case"]:
                raise RuntimeError("V4 ACK bytes differ from the prior same-case ACK")
            if not live_record or not live_record.get("controls"):
                raise RuntimeError("V4 live session has no committed control/ACK record")
            if live_record["controls"][-1].get("ack_sha256") != ack_report["sha256"]:
                raise RuntimeError("V4 ACK receipt differs from the live host control record")
            if not endpoint_comparison or endpoint_comparison.get("raw_bitwise_equal") is not True:
                raise RuntimeError("V4 full endpoint comparison did not pass")
        except BaseException as exc:
            error = exc
            error_traceback = traceback.format_exc()

    record = {
        "schema": "kdm6.native_upstream_nc_trace_v6_single_run_result.v1",
        "status": "FAILED" if error else "RUN_COMPLETE_FULL_COMPARISONS_PASS",
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "run_id": run_id,
        "plan_sha256": plan_receipt["plan_sha256"],
        "controller_sha256": sha256_file(Path(__file__)),
        "error": None if error is None else f"{type(error).__name__}: {error}",
        "traceback": error_traceback,
        "ready_comparison_path": str(output / "ready_comparison.json"),
        "ready_comparison_sha256": None if not ready_comparison else sha256_file(output / "ready_comparison.json"),
        "endpoint_comparison_path": str(output / "endpoint_comparison.json"),
        "endpoint_comparison_sha256": None if not endpoint_comparison else sha256_file(output / "endpoint_comparison.json"),
        "ack_comparison_path": str(output / "ack_comparison.json"),
        "ack_comparison_sha256": None if not ack_report else sha256_file(output / "ack_comparison.json"),
        "ack_sha256": None if not ack_report else ack_report["sha256"],
        "ack_bytes": None if not ack_report else ack_report["bytes"],
        "command_summary": command_summary,
        "trace_artifacts": trace_files,
        "trace_mode": case["trace_mode"],
        "upstream_nc_rows": len(upstream_rows),
        "live_session_record_path": str(live_record_path) if live_record is not None else None,
        "live_session_record_sha256": sha256_file(live_record_path) if live_record is not None else None,
        "native_host_steps_completed": 0 if live_record is None else sum(
            item.get("output_time") is not None for item in live_record.get("controls", [])),
        "output_time": None if not live_record or not live_record.get("controls") else live_record["controls"][-1].get("output_time"),
        "output_checkpoint_path": None if endpoint_path is None else str(endpoint_path),
        "output_checkpoint_sha256": None if not live_record or not live_record.get("controls") else live_record["controls"][-1].get("output_checkpoint_sha256"),
        "model_calls": {"host_steps": 1, "KDM6_wrapper_entries_expected": 2, "RTTOV": 0, "optimizer": 0},
    }
    result_path = output / "run_result.json"
    write_new(result_path, record)
    if result_path.stat().st_size > 64 * 1024:
        raise RuntimeError("V4 compact run result exceeded 64 KiB")
    if error is not None:
        raise error
    return record


def compare_v4_pairs(plan_path: Path) -> dict:
    plan, plan_receipt = _load_plan(plan_path)
    sys.path.insert(0, str(WORKTREE))
    from harness.native_host_transition import NativeCheckpoint
    pair_root = RUN_ROOT / "pair_comparisons"
    pair_root.mkdir(parents=True, exist_ok=True)
    pairs = []
    for case in ("control", "alpha025"):
        if case == "control":
            off_id = "control_upstream_nc_v4_off"
            off_dir = V4_CONTROL_OFF_DIR
            on_id = "control_upstream_nc_v6_on"
            on_dir = RUN_ROOT / on_id
        else:
            off_id = "alpha025_upstream_nc_v6_off"
            off_dir = RUN_ROOT / off_id
            on_id = "alpha025_upstream_nc_v6_on"
            on_dir = RUN_ROOT / on_id
        off = json.loads((off_dir / "run_result.json").read_text())
        on = json.loads((on_dir / "run_result.json").read_text())
        expected_status = "RUN_COMPLETE_FULL_COMPARISONS_PASS"
        if off.get("status") != expected_status or on.get("status") != expected_status:
            raise ValueError(f"{case} V4 off/on results are incomplete")
        if off.get("trace_mode") != "off" or on.get("trace_mode") != "on":
            raise ValueError(f"{case} off/on trace-mode labels are inconsistent")
        if off.get("output_checkpoint_sha256") != on.get("output_checkpoint_sha256"):
            raise ValueError(f"{case} V4 trace-on/off checkpoint hashes differ")
        if off.get("ack_sha256") != on.get("ack_sha256"):
            raise ValueError(f"{case} V4 trace-on/off ACK hashes differ")
        off_checkpoint = Path(off["output_checkpoint_path"])
        on_checkpoint = Path(on["output_checkpoint_path"])
        off_state = NativeCheckpoint.capture(off_checkpoint, expected_time="2025-07-19_00:01:00", expected_itimestep=3)
        comparison = _compare_full(off_checkpoint, off_state.file.sha256,
                                   NativeCheckpoint.capture(on_checkpoint, expected_time="2025-07-19_00:01:00", expected_itimestep=3))
        off_ack = json.loads((off_dir / "ack_comparison.json").read_text())
        on_ack = json.loads((on_dir / "ack_comparison.json").read_text())
        ack_equal = Path(off_ack["path"]).read_bytes() == Path(on_ack["path"]).read_bytes()
        if not ack_equal or comparison.get("raw_bitwise_equal") is not True:
            raise ValueError(f"{case} V4 trace-on/off full checkpoint or ACK differs")
        comparison_path = pair_root / f"{case}_trace_off_vs_on_full_checkpoint.json"
        comparison_sha = write_new(comparison_path, comparison)
        pair = {
            "schema": "kdm6.native_upstream_nc_trace_v6_noninterference_pair.v1",
            "case": case,
            "plan_sha256": plan_receipt["plan_sha256"],
            "trace_off_run_id": off_id,
            "trace_on_run_id": on_id,
            "final_checkpoint_sha256": off["output_checkpoint_sha256"],
            "raw_bitwise_equal": True,
            "full_checkpoint_comparison": {
                "path": str(comparison_path), "sha256": comparison_sha,
                "raw_bitwise_equal": comparison["raw_bitwise_equal"],
                "variable_count": comparison["variable_count"],
                "numeric_element_mismatches": comparison.get("numeric_element_mismatches"),
            },
            "ack_byte_identical": True,
            "ack_sha256": off_ack["sha256"],
            "upstream_nc_trace_sha256": on["trace_artifacts"]["upstream_nc.tsv"]["sha256"],
            "source_endpoint_comparisons_both_pass": True,
        }
        path = pair_root / f"{case}_trace_off_vs_on.json"
        pair["producer_sha256"] = sha256_file(Path(__file__))
        pair_sha = write_new(path, pair)
        pairs.append({"case": case, "path": str(path), "sha256": pair_sha,
                      "raw_bitwise_equal": True, "ack_byte_identical": True,
                      "final_checkpoint_sha256": off["output_checkpoint_sha256"]})
    result = {
        "schema": "kdm6.native_upstream_nc_trace_v6_noninterference_summary.v1",

def main(argv: list[str] | None = None) -> int:
    import argparse
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--plan", type=Path, default=PLAN_PATH)
    parser.add_argument("--run-id", choices=RUN_IDS)
    parser.add_argument("--authorization", type=Path)
    parser.add_argument("--compare-pairs", action="store_true")
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args(argv)
    if args.prepare:
        print(json.dumps(prepare_plan(), sort_keys=True))
        return 0
    if args.compare_pairs:
        print(json.dumps(compare_v4_pairs(args.plan), sort_keys=True))
        return 0
    if not args.run_id:
        parser.error("--run-id is required unless --prepare is used")
    if not args.execute:
        parser.error("--execute is required for a single host step")
    auth = args.authorization or AUTHORIZATION[args.run_id]
    print(json.dumps(execute_one(args.plan, args.run_id, auth), sort_keys=True))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
