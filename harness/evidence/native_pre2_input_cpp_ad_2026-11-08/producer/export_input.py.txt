#!/usr/bin/env python3
"""Export pinned NC16 native-state C ABI inputs for the graph-hook runner."""
from pathlib import Path
import hashlib
import numpy as np
import struct

PARENT = Path('/private/tmp/KDM6AD-pre2-cpp-ad/harness/evidence/native_nc16_vjp_branch_window_2026-11-06/raw/nc16_result.npz')
OUT = Path('host/research_evidence/native_pre2_input_ad_20261108/input.bin')
EXPECTED_SHA = 'a939fa0b11447da3c77d3fa67784ae5eb03fc5ccc49f42c5fb7eedb761443f57'
assert hashlib.sha256(PARENT.read_bytes()).hexdigest() == EXPECTED_SHA
with np.load(PARENT, allow_pickle=False) as z:
    s = z['state']; f = z['forcing']; land = z['xland']
    assert s.shape == (12,39) and s.dtype == np.float64
    assert f.shape == (4,39) and f.dtype == np.float64
    assert land.shape == (1,) and land.dtype == np.float32
    assert np.isfinite(s).all() and np.isfinite(f).all()
    raw = b'KDM6_PRE2_INPUT' + s.astype('<f8').tobytes() + f.astype('<f8').tobytes() + land.astype('<f4').tobytes() + struct.pack('<d',20.0)
OUT.write_bytes(raw)
print(len(raw),hashlib.sha256(raw).hexdigest())
