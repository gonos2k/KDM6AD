def file_sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(8 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()


def _write_json_new(path: Path, payload: dict) -> str:
    encoded = (
        json.dumps(payload, sort_keys=True, indent=2, allow_nan=False) + "\n"
    ).encode()
    with Path(path).open("xb") as stream:
        stream.write(encoded)
        stream.flush()
        os.fsync(stream.fileno())
    return hashlib.sha256(encoded).hexdigest()

def _load_trace_reader():
    spec = importlib.util.spec_from_file_location(
        "activation_trace_reader", TRACE_READER
    )
    if spec is None or spec.loader is None:
        raise RuntimeError("cannot load pinned activation trace reader")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

                        output = NativeBatchMicrophysics(
                            kernel,
                            geometry,
                            NativePrepConfig(**config["prep_configuration"]),
                            FinishConfig(**config["finish_configuration"]),
                            ZeroOutConfig(**config["zero_out_configuration"]),
                        )(theta, qv, mu, ph, rest)
                    if kernel.forward_calls - previous_forwards != 1:
                        raise RuntimeError(
                            "each FD endpoint must issue exactly one full-B KDM call"
                        )

                    native = {
                        name: numpy_copy(value)
                        for name, value in output.native_profile._asdict().items()
                    }
                    cloud = {
                        name: numpy_copy(value)
                        for name, value in output.cloud_state._asdict().items()
                    }
                    finish = np.stack(
                        [numpy_copy(value) for value in output.finish], axis=0
                    )
                    rho_d = numpy_copy(output.rho_d)
                    capture = dict(kernel.capture)
                    if not np.isfinite(capture["abi_raw_state_output"]).all():
                        raise ValueError(f"nonfinite raw KDM output at {tag}")

                    (_, _, _, _, _, _, cost_eligible, negative_nc, negative_ni, _) = (
                        _block_arrays(block)
                    )
                    candidates = np.flatnonzero(cost_eligible).astype(np.int64)
                    invalid = _validate_selected_inputs(
                        block, output, config, candidates
                    )
                    profile_valid = np.asarray(cost_eligible, dtype=bool).copy()
                    for bad_row in invalid:
                        profile_valid[int(bad_row)] = False
                    profile_rows = np.flatnonzero(profile_valid).astype(np.int64)
                    profile_products = {}
                    if profile_rows.size:
                        profiles = _selected_profile_context(
                            block, output, profile_rows, config
                        )
                        profile_products = {
                            "profile_" + name: np.array(value, copy=True)
                            for name, value in _profile_arrays(profiles).items()
                        }

                    products = {
                        "theta_m_input": theta_np,
                        "qv_input": qv_np,
                        "abi_state_input": capture["abi_state_input"],
                        "abi_forcing_input": capture["abi_forcing_input"],
                        "abi_raw_state_output": capture["abi_raw_state_output"],
                        "finish": finish,
                        "rho_d": rho_d,
                        "profile_valid": profile_valid,
                        "profile_rows": profile_rows,
                        "profile_invalid_rows": np.asarray(
                            sorted(invalid), dtype=np.int32
                        ),
                        "initial_cost_eligible": cost_eligible,
                        "initial_negative_nc": negative_nc,
                        "initial_negative_ni": negative_ni,
                        "source_dqf": source_arrays["source_dqf"],
                        "source_qc": source_arrays["source_qc"],
                        "source_eligible": source_arrays["source_eligible"],
                        "baseline_quality": baseline_support["quality"],
                        "baseline_mask": baseline_support["mask"],
                        "baseline_weights": baseline_support["weights"],
                        "baseline_profile_valid": baseline_support["profile_valid"],
                        "baseline_profile_rows": baseline_support["profile_rows"],
                        "baseline_target_CFRAC": baseline_target_cfrac,
                    }
                    products.update(
                        {"native_" + name: value for name, value in native.items()}
                    )
                    products.update(
                        {"cloud_" + name: value for name, value in cloud.items()}
                    )
                    products.update(profile_products)
                    products_path = case_output / "endpoint.npz"
                    with products_path.open("xb") as stream:
                        np.savez_compressed(stream, **products)
                        stream.flush()
                        os.fsync(stream.fileno())
                    endpoint_receipt = {
                        "schema": "kdm6.block22_centered_fd_endpoint.v1",
                        "status": "KDM_AND_PROFILE_BUILDER_COMPLETE",
                        "epsilon": epsilon,
                        "sign": sign,
                        "tag": tag,
                        "trace_call_ordinal": 1
                        + 2 * eps_index
                        + (0 if sign > 0 else 1),
                        "fullB_KDM_calls": 1,
                        "RTTOV_calls": 0,
                        "profile_builder_calls": int(profile_rows.size > 0),
                        "quality_status": "RTTOV quality not evaluated; saved baseline fixed support and source QC/DQF",
                        "invalid_profile_rows": {
                            str(key): value for key, value in invalid.items()
                        },
                        "endpoint_npz": str(products_path),
                        "endpoint_npz_sha256": file_sha256(products_path),
                        "trace_path": str(trace_path),
                        "trace_source_receipt": str(OBSERVER_SOURCE_RECEIPT),
                    }
                    _write_json_new(case_output / "receipt.json", endpoint_receipt)
                    endpoints.append(endpoint_receipt)
                    print(
                        json.dumps(
                            {
                                "endpoint": tag,
                                "status": endpoint_receipt["status"],
                                "npz_sha256": endpoint_receipt["endpoint_npz_sha256"],
                            }
                        )
                    )
    except Exception:
        _write_json_new(
            output_dir / "execution_failure.json",
            {
                "schema": "kdm6.block22_centered_fd_failure.v1",
                "status": "FAILED",
                "failed_at_utc": datetime.now(timezone.utc).isoformat(),
                "completed_endpoints": endpoints,
                "trace_path": str(trace_path),
                "exception": traceback.format_exc(),
            },
        )
        raise

    if (
        len(endpoints) != 10
        or base_theta.tobytes() != base_theta_bytes
        or base_qv.tobytes() != base_qv_bytes
    ):
        raise RuntimeError(
            "endpoint coverage or baseline-input immutability check failed"
        )

    parsed_trace = trace_reader.read_trace(trace_path, OBSERVER_SOURCE_RECEIPT)
    if sorted(parsed_trace["calls"]) != list(range(1, 11)):
        raise ValueError(
            "activation trace must contain exactly ten ordered endpoint calls"
        )
    baseline_trace = trace_reader.read_trace(AD_TRACE, OBSERVER_SOURCE_RECEIPT)
    if 1 not in baseline_trace["calls"]:
        raise ValueError("trace-on AD baseline has no first call to compare")
    branch_comparisons = []
    for endpoint in endpoints:
        ordinal = endpoint["trace_call_ordinal"]
        comparison = trace_reader.compare_trace_calls(
            baseline_trace, 1, parsed_trace, ordinal
        )
        mask_changes = comparison["calls"][0]["mask_changes"]
        target_changes = [
            {
                "stage": item["stage"],
                "field": item["field"],
                "changed_target_levels": len(item["target_row_changed_levels"]),
                "base_true_target_levels": item["base_true_target_levels"],
                "probe_true_target_levels": item["probe_true_target_levels"],
            }
            for item in mask_changes
            if item["target_row_changed_levels"]
        ]
        comparison["endpoint"] = endpoint["tag"]
        comparison["target_row0"] = TARGET_ROW
        comparison["target_mask_changes"] = target_changes
        branch_comparisons.append(comparison)
        endpoint["target_mask_change_count"] = int(
            sum(item["changed_target_levels"] for item in target_changes)
        )

    endpoint_arrays = {}
    for endpoint in endpoints:
        with np.load(endpoint["endpoint_npz"], allow_pickle=False) as archive:
            endpoint_arrays[endpoint["tag"]] = {
                key: archive[key] for key in archive.files
            }
        sample = endpoint_arrays[endpoint["tag"]]
        if not same_bits(
            sample["abi_state_input"][2:], ad["baseline__abi_state_input"][2:]
        ):
            raise ValueError(f"fixed state rows changed at {endpoint['tag']}")
        if not same_bits(sample["abi_state_input"][1], sample["qv_input"]):
            raise ValueError(
                f"QV ABI state differs from endpoint input at {endpoint['tag']}"
            )
        rows = sample["profile_rows"]
        if rows.ndim != 1 or np.unique(rows).size != rows.size:
            raise ValueError(f"profile row ownership is invalid at {endpoint['tag']}")
    per_epsilon = []
    experiment = json.loads(EXPERIMENT_PLAN.read_text())
    profile_atol = float(experiment["local_FD"]["profile_coordinate_atol"])
    profile_rtol = float(experiment["local_FD"]["profile_coordinate_rtol"])
    for epsilon in EPSILONS:
        plus_tag, minus_tag = f"eps_{epsilon:.0e}_plus", f"eps_{epsilon:.0e}_minus"
        plus, minus = endpoint_arrays[plus_tag], endpoint_arrays[minus_tag]
        raw_fd = (plus["abi_raw_state_output"] - minus["abi_raw_state_output"]) / (
            2.0 * epsilon
        )
        finish_fd = (plus["finish"] - minus["finish"]) / (2.0 * epsilon)
        forcing_fd = (plus["abi_forcing_input"] - minus["abi_forcing_input"]) / (
            2.0 * epsilon
        )
        forcing_ref = ad["baseline__abi_forcing_direction"]
        forcing_signal = np.maximum(np.abs(forcing_fd), np.abs(forcing_ref))
        forcing_ratio_by_field = np.max(
            np.abs(forcing_fd - forcing_ref) / (1.0e-8 + 3.0e-5 * forcing_signal),
            axis=(1, 2),
        )
        forcing_prep_pass = bool(np.all(forcing_ratio_by_field <= 1.0))
        raw_ref = ad["baseline__abi_raw_state_jvp"]
        finish_ref = ad["baseline__tangent_finish"]
        fd17 = np.concatenate((raw_fd, finish_fd), axis=0)
        ref17 = np.concatenate((raw_ref, finish_ref), axis=0)
        scale17 = np.asarray(SCALES, dtype=np.float64).reshape(17, 1, 1)
        atol17 = np.asarray(ATOL, dtype=np.float64).reshape(17, 1, 1)
        rtol17 = np.asarray(RTOL, dtype=np.float64).reshape(17, 1, 1)
        scaled_error = np.abs((fd17 - ref17) / scale17)
        scaled_signal = np.maximum(np.abs(fd17 / scale17), np.abs(ref17 / scale17))
        allB_ratio_by_output = np.max(scaled_error, axis=(1, 2)) / (
            atol17[:, 0, 0] + rtol17[:, 0, 0] * np.max(scaled_signal, axis=(1, 2))
        )
        target_err = np.max(scaled_error[:, TARGET_ROW, :], axis=1)
        target_signal = np.max(scaled_signal[:, TARGET_ROW, :], axis=1)
        target_ratio_by_output = target_err / (
            atol17[:, 0, 0] + rtol17[:, 0, 0] * target_signal
        )
        target_pass = bool(np.all(target_ratio_by_output <= 1.0))
        plus_pos = np.flatnonzero(plus["profile_rows"] == TARGET_ROW)
        minus_pos = np.flatnonzero(minus["profile_rows"] == TARGET_ROW)
        profile_valid = bool(
            plus["profile_valid"][TARGET_ROW]
            and minus["profile_valid"][TARGET_ROW]
            and plus_pos.size == 1
            and minus_pos.size == 1
        )
        profile_results = {}
        profile_fd_values = {}
        if profile_valid:
            for field in experiment["local_FD"]["profile_fields"]:
                plus_profile = plus[f"profile_{field}"][int(plus_pos[0])]
                minus_profile = minus[f"profile_{field}"][int(minus_pos[0])]
                profile_fd = (plus_profile - minus_profile) / (2.0 * epsilon)
                profile_tangent = profile_jvp[f"JVP_{field}"][0]
                if profile_fd.shape != profile_tangent.shape:
                    raise ValueError(f"profile FD shape mismatch for {field}")
                tolerance = profile_atol + profile_rtol * np.maximum(
                    np.abs(profile_fd), np.abs(profile_tangent)
                )
                coordinate_ratio = np.abs(profile_fd - profile_tangent) / tolerance
                field_max = float(np.max(coordinate_ratio))
                profile_results[field] = {
                    "max_coordinate_tolerance_ratio": field_max,
                    "coordinate_count": int(coordinate_ratio.size),
                    "passed": field_max <= 1.0,
                }
                profile_fd_values[field] = profile_fd
        profile_pass = bool(
            profile_valid
            and len(profile_results) == len(experiment["local_FD"]["profile_fields"])
            and all(item["passed"] for item in profile_results.values())
        )
        profile_fd_values["CFRAC_plus"] = (
            plus["profile_CFRAC"][int(plus_pos[0])] if profile_valid else np.array([])
        )
        profile_fd_values["CFRAC_minus"] = (
            minus["profile_CFRAC"][int(minus_pos[0])] if profile_valid else np.array([])
        )
        profile_fd_values["CFRAC_baseline"] = plus["baseline_target_CFRAC"]
        cfrac_plus_changes = (
            int(
                np.count_nonzero(
                    profile_fd_values["CFRAC_plus"]
                    != profile_fd_values["CFRAC_baseline"]
                )
            )
            if profile_valid
            else None
        )
        cfrac_minus_changes = (
            int(
                np.count_nonzero(
                    profile_fd_values["CFRAC_minus"]
                    != profile_fd_values["CFRAC_baseline"]
                )
            )
            if profile_valid
            else None
        )
        cfrac_exact_both = bool(
            profile_valid
            and same_bits(
                profile_fd_values["CFRAC_plus"], profile_fd_values["CFRAC_baseline"]
            )
            and same_bits(
                profile_fd_values["CFRAC_minus"], profile_fd_values["CFRAC_baseline"]
            )
        )
        profile_common = target_pass and profile_pass and cfrac_exact_both
        per_epsilon.append(
            {
                "epsilon": epsilon,
                "target_row0": TARGET_ROW,
                "target_output_max_ratio": float(np.max(target_ratio_by_output)),
                "target_pass_all_17_outputs": target_pass,
                "target_ratio_by_output": target_ratio_by_output.tolist(),
                "forcing_input_JVP_diagnostic_max_ratio": float(
                    np.max(forcing_ratio_by_field)
                ),
                "forcing_input_JVP_diagnostic_ratio_by_field": forcing_ratio_by_field.tolist(),
                "forcing_input_JVP_diagnostic_pass": forcing_prep_pass,
                "profile_valid_at_both_endpoints": profile_valid,
                "profile_pass_all_8_fields": profile_pass,
                "profile_max_ratio": max(
                    (
                        item["max_coordinate_tolerance_ratio"]
                        for item in profile_results.values()
                    ),
                    default=None,
                ),
                "profile_by_field": profile_results,
                "target_profile_CFRAC_plus_changed_coordinates_vs_baseline": cfrac_plus_changes,
                "target_profile_CFRAC_minus_changed_coordinates_vs_baseline": cfrac_minus_changes,
                "target_profile_CFRAC_exact_bits_at_both_endpoints": cfrac_exact_both,
                "target_mask_change_count_plus": next(
                    item["target_mask_change_count"]
                    for item in endpoints
                    if item["tag"] == plus_tag
                ),
                "target_mask_change_count_minus": next(
                    item["target_mask_change_count"]
                    for item in endpoints
                    if item["tag"] == minus_tag
                ),
                "fullB_profile_valid_changed_rows_plus": int(
                    np.count_nonzero(
                        plus["profile_valid"] != baseline_support["profile_valid"]
                    )
                ),
                "fullB_profile_valid_changed_rows_minus": int(
                    np.count_nonzero(
                        minus["profile_valid"] != baseline_support["profile_valid"]
                    )
                ),
                "target_state_finish_profile_common_pass": profile_common,
                "fullB_max_ratio": float(np.max(allB_ratio_by_output)),
                "fullB_ratio_by_output": allB_ratio_by_output.tolist(),
                "raw12_fd_fullB": raw_fd,
                "finish5_fd_fullB": finish_fd,
                "forcing4_fd_fullB": forcing_fd,
                "profile_fd_values": profile_fd_values,
            }
        )
        del (
            fd17,
            ref17,
            scale17,
            atol17,
            rtol17,
            scaled_error,
            scaled_signal,
            forcing_signal,
        )

    best_common = [
        item["epsilon"]
        for item in per_epsilon
        if item["target_state_finish_profile_common_pass"]
    ]
    summary = {
        "schema": "kdm6.block22_centered_fd_result.v2",
        "status": "COMPLETE_ALL_ENDPOINTS",
        "completed_at_utc": datetime.now(timezone.utc).isoformat(),
        "elapsed_seconds": time.monotonic() - start_time,
        "launch_plan_sha256": file_sha256(plan_path),
        "runner_sha256": file_sha256(Path(__file__)),
        "endpoint_count": len(endpoints),
        "fullB_KDM_calls": len(endpoints),
        "RTTOV_calls": 0,
        "branch_and_schedule_comparison_scope": "full-B512 captured boolean masks and schedule vectors vs AD trace-on baseline KDM call 1; not complete internal branch proof",
        "branch_and_schedule_comparisons": branch_comparisons,
        "target_primary_scope": "row 368: raw ABI state 12 plus native finish 5 at 39 levels and eight differentiable RTTOV profile fields at 66/67 levels; all 25 KDM/profile categories must pass at one common epsilon. BT7 and Huber cost remain pending direct RTTOV.",
        "prior_orphaned_KDM6_calls": 1,
        "total_campaign_KDM6_calls_including_orphan": 1 + len(endpoints),
        "profile_tolerances": {
            "absolute": profile_atol,
            "relative": profile_rtol,
            "units": experiment["local_FD"]["profile_units"],
        },
        "forcing_input_JVP_diagnostic_only": {
            "absolute_per_coordinate": 1.0e-8,
            "relative_per_coordinate": 3.0e-5,
            "envelope": "abs(FD-JVP)/(1e-8+3e-5*max(abs(FD),abs(JVP)))",
            "scope": "DEN, PII, P, DELZ over B512/L39; reported only and excluded from the 33-category acceptance gate",
        },
        "tolerances_imported": {
            "source": str(SOURCE_V2 / "harness/verify_localized_ice_ad.py"),
            "state_atol": float(ATOL[0]),
            "state_rtol": float(RTOL[0]),
            "finish_atol": float(ATOL[12]),
            "finish_rtol": float(RTOL[12]),
        },
        "target_state_finish_profile_common_passing_epsilons": best_common,
        "target_state_finish_profile_common_epsilon_passed": bool(best_common),
        "full_33_category_acceptance_complete": False,
        "observation_status": "No endpoint RTTOV was run; target BT/cost FD pending separate reviewed pilot/controller",
        "per_epsilon": [
            {
                key: value
                for key, value in item.items()
                if key
                not in (
                    "raw12_fd_fullB",
                    "finish5_fd_fullB",
                    "forcing4_fd_fullB",
                    "profile_fd_values",
                )
            }
            for item in per_epsilon
        ],
        "endpoints": endpoints,
    }
    _write_json_new(output_dir / "centered_fd_summary.json", summary)
    fd_bundle = {}
    for item in per_epsilon:
        eps_tag = f"{item['epsilon']:.0e}"
        fd_bundle[f"eps_{eps_tag}__raw12_fd_fullB"] = item["raw12_fd_fullB"]
        fd_bundle[f"eps_{eps_tag}__finish5_fd_fullB"] = item["finish5_fd_fullB"]
        fd_bundle[f"eps_{eps_tag}__forcing4_fd_fullB"] = item["forcing4_fd_fullB"]
        for field, value in item["profile_fd_values"].items():
            fd_bundle[f"eps_{eps_tag}__target_profile_{field}"] = value
    bundle_path = output_dir / "centered_fd_fullB.npz"
    with bundle_path.open("xb") as stream:
        np.savez_compressed(stream, **fd_bundle)
        stream.flush()
        os.fsync(stream.fileno())
    _write_json_new(
        output_dir / "execution_complete.json",
        {
            "schema": "kdm6.block22_centered_fd_execution.v1",
            "status": "COMPLETE_ALL_ENDPOINTS",
            "completed_at_utc": datetime.now(timezone.utc).isoformat(),
            "launch_plan_sha256": file_sha256(plan_path),
            "summary_sha256": file_sha256(output_dir / "centered_fd_summary.json"),
            "fullB_fd_bundle": str(bundle_path),
            "fullB_fd_bundle_sha256": file_sha256(bundle_path),
            "endpoint_count": len(endpoints),
            "fullB_KDM_calls": len(endpoints),
            "prior_orphaned_KDM6_calls": 1,
            "total_campaign_KDM6_calls_including_orphan": 1 + len(endpoints),
            "RTTOV_calls": 0,
        },
    )
    print(
        json.dumps(
            {
                "status": summary["status"],
                "endpoints": len(endpoints),
                "target_state_finish_profile_common_passing_epsilons": best_common,
                "summary": str(output_dir / "centered_fd_summary.json"),
            }
        )
    )
    return 0 if best_common else 2


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    action = parser.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--execute", action="store_true")
    parser.add_argument("--output", required=True, type=Path)
    args = parser.parse_args(argv)
    if args.prepare:
        _prepare(args.output)
        return 0
    return _execute(args.output)


if __name__ == "__main__":
    raise SystemExit(main())
