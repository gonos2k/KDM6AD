"""Isolated one-TU QR candidate build against the source-pinned frozen v6 core."""
from pathlib import Path
import hashlib
import json
import shlex
import shutil
import subprocess

ROOT = Path(__file__).resolve().parent
REPO = Path('/private/tmp/KDM6AD-actual-qr-20261024')
BASE = Path('/Users/yhlee/KDM6AD-k/host/research_evidence/mp137_ice_factorial_20261023/overlay_build_v6/factorial_build/libkdm6.a')
OLD = Path('/Users/yhlee/KDM6AD-k/host/research_evidence/mp137_same_call_budget_20261022/build')
BATCH = Path('/Users/yhlee/KDM6AD-k/host/research_evidence/mp137_ice_factorial_20261023/batch_tool_build/ice_factorial_batch.o')
API = Path('/Users/yhlee/KDM6AD-k/host/research_evidence/mp137_ice_boundary_20261023/forensic_c/kdm6_c_api.cpp.o')
BUILD = ROOT/'build_final'
BUILD.mkdir(exist_ok=False)


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def save(path, data):
    with path.open('x') as stream:
        json.dump(data, stream, indent=2)
        stream.write('\n')


assert sha(BASE) == '0bf2c94d30d66ba5a76767ae526ee5ea89d944a8cc242f95255b20a23bdb8b3a'
assert sha(API) == 'a95211b6186982780c1e54bc76dffa1db159bb8f4929679d093f32b5269e95a0'
flags = {}
for line in (OLD/'CMakeFiles/kdm6.dir/flags.make').read_text().splitlines():
    if ' = ' in line:
        key, value = line.split(' = ', 1)
        flags[key] = shlex.split(value)
includes = [('-I'+str(REPO/'libtorch/include')) if token.startswith('-I') and token.endswith('/libtorch/include') else token for token in flags['CXX_INCLUDES']]
cxx = '/Applications/Xcode.app/Contents/Developer/Toolchains/XcodeDefault.xctoolchain/usr/bin/c++'
command = [cxx, *flags['CXX_DEFINES'], *includes, *flags['CXX_FLAGS'], '-c', str(ROOT/'overlay_final/sedimentation.cpp'), '-o', str(BUILD/'sedimentation.cpp.o')]
assert '-ffp-contract=off' in flags['CXX_FLAGS']
inputs = [BASE, BATCH, API, Path(__file__), OLD/'CMakeFiles/kdm6.dir/flags.make', ROOT/'overlay_final/source_sha256.json', REPO/'harness/make_actual_qr_overlay.py', REPO/'oracle/kdm6/qr_actual_transport.py']
inputs += list((ROOT/'overlay_final').glob('*.cpp')) + list((REPO/'libtorch/include').rglob('*.h'))
pins = {str(path): sha(path) for path in inputs}
save(BUILD/'prepared.json', {'source_sha256':pins, 'compile_command':command, 'scope':'one production TU replaced; original flags retained; no host/install or test-target change'})
with (BUILD/'compile.log').open('xb') as log:
    subprocess.run(command, stdout=log, stderr=subprocess.STDOUT, check=True)
archive = BUILD/'libkdm6_actual_qr.a'
shutil.copyfile(BASE, archive)
subprocess.run(['/usr/bin/ar', '-r', str(archive), str(BUILD/'sedimentation.cpp.o')], check=True)
subprocess.run(['/usr/bin/ranlib', str(archive)], check=True)
before = subprocess.check_output(['/usr/bin/ar', '-t', str(BASE)], text=True).splitlines()
after = subprocess.check_output(['/usr/bin/ar', '-t', str(archive)], text=True).splitlines()
assert len(before) == len(set(before)) and set(before) == set(after)
objects = []
for name in before:
    old = subprocess.check_output(['/usr/bin/ar', '-p', str(BASE), name])
    new = subprocess.check_output(['/usr/bin/ar', '-p', str(archive), name])
    objects.append({'name':name,'before_sha256':hashlib.sha256(old).hexdigest(),'after_sha256':hashlib.sha256(new).hexdigest(),'identical':old == new})
changed = [row['name'] for row in objects if not row['identical'] and row['name'].endswith('.o')]
assert changed == ['sedimentation.cpp.o']
links = []
for target in ['kdm6_column_budget', 'kdm6_c']:
    cmd = shlex.split((OLD/f'CMakeFiles/{target}.dir/link.txt').read_text())
    for index, value in enumerate(cmd):
        if value == 'CMakeFiles/kdm6_column_budget.dir/tools/kdm6_column_budget.cpp.o':cmd[index] = str(BATCH)
        elif value == 'CMakeFiles/kdm6_c.dir/bridge/kdm6_c_api.cpp.o':cmd[index] = str(API)
        elif value == 'kdm6_column_budget':cmd[index] = str(BUILD/'actual_qr_batch')
        elif value == 'libkdm6_c.2.0.0.dylib':cmd[index] = str(BUILD/'libkdm6_actual_qr.2.0.0.dylib')
        elif value == 'libkdm6.a':cmd[index] = str(archive)
        elif value.startswith('-Wl,-exported_symbols_list,'):cmd[index] = '-Wl,-exported_symbols_list,'+str(REPO/'libtorch/bridge/kdm6_c.exports')
        elif value == '@rpath/libkdm6_c.2.dylib':cmd[index] = '@rpath/libkdm6_actual_qr.2.dylib'
    with (BUILD/f'link_{target}.log').open('xb') as log:
        subprocess.run(cmd, stdout=log, stderr=subprocess.STDOUT, check=True)
    links.append(cmd)
assert all(sha(path) == digest for path,digest in pins.items())
save(BUILD/'receipt.json', {'source_sha256':pins,'objects':objects,'changed_object_members':changed,'unchanged_object_members':13,'archive_symbol_index_separate':True,'link_commands':links,'archive_sha256':sha(archive),'batch_tool_sha256':sha(BUILD/'actual_qr_batch'),'library_sha256':sha(BUILD/'libkdm6_actual_qr.2.0.0.dylib'),'operational_install_changed':False})
print('candidate build complete', flush=True)
