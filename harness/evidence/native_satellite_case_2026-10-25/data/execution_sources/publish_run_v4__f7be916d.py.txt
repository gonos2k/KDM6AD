"""Prepare a small, path-redacted evidence pack from a completed RUN_v4.

Private evidence tooling only. This script reads the final run and independent
audits; it never executes KDM, RTTOV, the host, or the assimilation. Its output
directory is exclusive and is not created until all completion and audit gates
have passed.
"""

RUN_DEFAULT = HERE / "RUN_v4"
OUT_DEFAULT = RUN_DEFAULT / "PUBLICATION_PACK_FINAL"
NATIVE_COLUMNS = 65_988
NATIVE_LEVELS = 39
CHANNELS = (10, 11, 12, 13, 14, 15, 16)
CHANNEL_NAMES = ("wv073", "ir087", "ir096", "ir105", "ir112", "ir123", "ir133")
MAX_NPZ_BYTES = 50 * 1024 * 1024


def _sha_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def _sha_file(path: Path) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def _json(path: Path):
    try:
        return json.loads(Path(path).read_text())
    except (OSError, json.JSONDecodeError) as exc:
        raise ValueError(f"cannot read valid JSON: {path.name}") from exc


def _redact_paths(value):
    if isinstance(value, dict):
        return {key: _redact_paths(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)):
        return [_redact_paths(item) for item in value]
    if isinstance(value, str):
        return re.sub(r"/(?:Users|private)/[^\s\"'`,;)]*", "[private path redacted]", value)
    return value


def _write_json(path: Path, value) -> None:
    with Path(path).open("x") as stream:
        json.dump(_redact_paths(value), stream, indent=2, sort_keys=True, allow_nan=False)
        stream.write("\n")


def _identity(value) -> str:
    payload = json.dumps(value, sort_keys=True, allow_nan=False, separators=(",", ":"))
    return _sha_bytes(payload.encode())


def _inside(root: Path, path: Path, label: str) -> Path:
    root = Path(root).resolve(strict=True)
    path = Path(path)
    if path.is_symlink():
        raise ValueError(f"{label} must not be a symlink")
    resolved = path.resolve(strict=True)
    if resolved == root or root not in resolved.parents:
        raise ValueError(f"{label} is outside the run root")
    return resolved


def _finite_float(value, name: str) -> float:
    result = float(value)
    if not math.isfinite(result):
        raise ValueError(f"{name} must be finite")
    return result


def _check_baseline(run: Path, run_id: str, case_sha: str):
    baseline_path = run / "BASELINE.json"
    baseline = _json(baseline_path)
    entries = baseline.get("chunks")
    if (baseline.get("full_domain") is not True or baseline.get("processed_columns") != NATIVE_COLUMNS
            or not isinstance(entries, list) or len(entries) != 129):
        raise ValueError("BASELINE.json is not a complete 129-block native-domain result")
    ordered = sorted(entries, key=lambda row: row.get("chunk_index", -1))
    cursor = 0
    for index, row in enumerate(ordered):
        count = 512 if index < 128 else 452
        if (row.get("chunk_index") != index or row.get("start") != cursor
                or row.get("stop") != cursor + count or row.get("columns") != count):
            raise ValueError(f"baseline block {index} has a missing, duplicate, or wrong range")
        cursor += count
        receipt_path = run / "baseline" / f"chunk_{index:05d}" / "receipt.json"
        receipt = _json(receipt_path)
        if receipt != row:
            raise ValueError(f"baseline block {index} manifest/receipt mismatch")
        binding_data = {
            "run_id": run_id, "stage": "baseline", "chunk": index,
            "baseline_sha": None, "controls_sha": None,
        }
        binding = _sha_bytes(json.dumps(
            binding_data, sort_keys=True, allow_nan=False, separators=(",", ":")
        ).encode())
        if receipt.get("binding") != binding:
            raise ValueError(f"baseline block {index} binding mismatch")
        artifact = _inside(run, Path(receipt["artifact"]), f"baseline block {index} artifact")
        if artifact.parent.parent.resolve() != receipt_path.parent.resolve():
            raise ValueError(f"baseline block {index} artifact escaped its receipt directory")
        if _sha_file(artifact) != receipt.get("artifact_sha256"):
            raise ValueError(f"baseline block {index} artifact SHA mismatch")
        if receipt.get("calls", {}).get("KDM_forward") != 1:
            raise ValueError(f"baseline block {index} lacks its KDM forward receipt")
    if cursor != NATIVE_COLUMNS:
        raise ValueError("baseline ranges do not cover 65,988 native columns")
    z = _finite_float(baseline.get("globalZ"), "baseline globalZ")
    if z <= 0.0:
        raise ValueError("baseline globalZ must be positive")
    dot = baseline.get("dot", {})
    if dot.get("passed") is not True or _finite_float(dot.get("relative_error"), "baseline dot error") > 1e-10:
        raise ValueError("baseline global JVP/VJP gate did not pass")
    if sum(row.get("columns", 0) for row in ordered) != NATIVE_COLUMNS:
        raise ValueError("baseline receipt columns do not sum to the native grid")
    if baseline.get("call_totals", {}).get("KDM_forward") != 129:
        raise ValueError("baseline must record exactly one full-B KDM forward per native block")
    return baseline, ordered


def _check_final(run: Path, run_id: str, baseline):
    final_path = run / "FINAL.json"
    decision_path = run / "RUN_DECISION.json"
    receipt_path = run / "SUMMARY_RECEIPT.json"
    if not (final_path.is_file() and decision_path.is_file() and receipt_path.is_file()):
        raise ValueError("publication requires FINAL, RUN_DECISION and SUMMARY_RECEIPT")
    final, decision, summary_receipt = map(_json, (final_path, decision_path, receipt_path))
    if final.get("status") not in ("COMPLETED_ACCEPTED", "COMPLETED_NO_ACCEPTABLE_UPDATE"):
        raise ValueError("FINAL status is not a completed run")
    if final.get("processed_columns") != NATIVE_COLUMNS or final.get("blocks") != 129:
        raise ValueError("FINAL does not declare full-domain coverage")
    if final.get("native_host_steps") != 0 or final.get("native_cycles") != 0:
        raise ValueError("this offline case must not claim a host step or cycle")
    for name in ("status", "accepted_trial", "trials", "processed_columns", "blocks"):
        if final.get(name) != decision.get(name):
            raise ValueError(f"FINAL and RUN_DECISION differ in {name}")
    if _identity(final) != _identity(decision):
        raise ValueError("FINAL and RUN_DECISION are not the same full decision object")
    if final.get("gradient_verification", {}).get("dot", {}).get("passed") is not True:
        raise ValueError("FINAL lacks a passing derivative dot gate")
    if final.get("gradient_verification", {}).get("sparse_FD_passed") is not True:
        raise ValueError("FINAL lacks a passing independent sparse FD gate")
    if final.get("status") == "COMPLETED_ACCEPTED":
        accepted = final.get("accepted_trial")
        if not accepted or accepted not in final.get("trials", []):
            raise ValueError("accepted FINAL must identify one declared trial")
    elif final.get("accepted_trial") is not None:
        raise ValueError("no-acceptable-update FINAL must not identify a trial")
    if baseline.get("processed_columns") != final["processed_columns"]:
        raise ValueError("BASELINE and FINAL coverage differ")
    if not isinstance(summary_receipt.get("decision_identity"), str):
        raise ValueError("SUMMARY_RECEIPT lacks its decision identity")
    if summary_receipt["decision_identity"] != _identity(final):
        raise ValueError("SUMMARY_RECEIPT decision identity differs from FINAL")
    summary_dir = _inside(run, Path(summary_receipt["directory"]), "summary directory")
    if summary_dir.parent.resolve() != run.resolve():
        raise ValueError("summary directory must be directly under RUN_v4")
    files = summary_receipt.get("files_sha256")
    if not isinstance(files, dict) or "assembled.npz" not in files or "summary.json" not in files:
        raise ValueError("SUMMARY_RECEIPT lacks assembled arrays or summary metadata")
    for name, expected in files.items():
        path = summary_dir / name
        if path.parent.resolve() != summary_dir or not path.is_file() or _sha_file(path) != expected:
            raise ValueError(f"summary artifact failed its receipt: {name}")
    summary = _json(summary_dir / "summary.json")
    if summary.get("status") != final.get("status") or summary.get("grid", {}).get("profiles") != NATIVE_COLUMNS:
        raise ValueError("summary metadata differs from FINAL/domain")
    return final, summary, summary_dir, files, final_path, summary_receipt


def _check_input_audits(run: Path, run_id: str, case):
    native = _json(run / "ALL_NATIVE_INPUT_BLOCKS_AUDIT.json")
    if (native.get("source_run_id") != run_id or native.get("columns") != NATIVE_COLUMNS
            or native.get("blocks") != 129 or native.get("first_flat") != 0
            or native.get("last_flat") != NATIVE_COLUMNS - 1
            or native.get("source_QC_zero_observations") != NATIVE_COLUMNS * len(CHANNELS)
            or len(native.get("records", [])) != 129
            or _finite_float(native.get("max_prep_abs_vs_original_packet"), "all-native prep replay") > 1e-10):
        raise ValueError("ALL_NATIVE_INPUT_BLOCKS_AUDIT does not certify the full case input")
    native_records = sorted(native["records"], key=lambda row: row.get("chunk", -1))
    cursor = 0
    for i, record in enumerate(native_records):
        count = 512 if i < 128 else 452
        if (record.get("chunk") != i or record.get("start") != cursor
                or record.get("stop") != cursor + count or record.get("columns") != count):
            raise ValueError(f"all-native input audit range mismatch at chunk {i}")
        cursor += count
    if cursor != NATIVE_COLUMNS:
        raise ValueError("all-native input audit has incomplete ranges")

    raw = _json(run / "RAW16_QUALITY_AND_FOOTPRINT_AUDIT.json")
    if (raw.get("source_run_id") != run_id or raw.get("raw_channels") != 16
            or len(raw.get("records", [])) != 16 or raw.get("no_KDM_calls") is not True
            or raw.get("no_RTTOV_calls") is not True
            or raw.get("no_model_or_observation_modification") is not True):
        raise ValueError("RAW16 quality/footprint audit identity or scope mismatch")
    records = raw["records"]
    by_channel = {row.get("channel"): row for row in records}
    inventory = {row["channel"]: row for row in case["case_preparation"]["raw16_channel_inventory"]}
    if set(by_channel) != set(inventory):
        raise ValueError("raw-channel audit differs from the 16-source inventory")
    selected = {"wv073", "ir087", "ir096", "ir105", "ir112", "ir123", "ir133"}
    for name, row in by_channel.items():
        if row.get("source_sha256") != inventory[name].get("sha256") or row.get("source_bytes_unchanged") is not True:
            raise ValueError(f"raw channel {name} changed or is not pinned")
        if name in selected and (
            row.get("selected_for_cost") is not True
            or row.get("model_centers_total") != NATIVE_COLUMNS
            or row.get("model_centers_outside") != 0
            or row.get("mapped_missing_pixels") != 0
            or row.get("mapped_quality0_nonmissing") != NATIVE_COLUMNS
        ):
            raise ValueError(f"selected raw channel {name} lacks full quality/footprint coverage")
    if raw.get("pixel_UTC_verified") is not False or raw.get("provider_original_authenticity_certified") is not False:
        raise ValueError("raw observation provenance caveats are absent")
    return native, raw


def _check_owner_audit(run: Path, run_id: str, case):
    owner_path = run / "KO_NEAREST_OWNER_AUDIT.json"
    owner = _json(owner_path)
    if (owner.get("schema") != "kdm6.native_KO_nearest_owner_audit.v1"
            or owner.get("run_id") != run_id or owner.get("passed") is not True
            or owner.get("native_columns") != NATIVE_COLUMNS
            or owner.get("candidate_KO_pixels") != 900 * 900
            or owner.get("pixel_mismatches") != 0
            or owner.get("unique_saved_pixel_owners") != NATIVE_COLUMNS
            or owner.get("unique_recomputed_pixel_owners") != NATIVE_COLUMNS
            or not all(item.get("matches_tree_and_saved") is True
                       for item in owner.get("direct_exhaustive_samples", []))
            or len(owner.get("direct_exhaustive_samples", [])) != 14
            or owner.get("new_model_or_RTTOV_calls") != 0
            or owner.get("quality_used_to_choose_owner") is not False
            or owner.get("provider_original_authenticity_certified") is not False
            or owner.get("pixel_UTC_verified") is not False):
        raise ValueError("nearest-owner audit does not certify the full unchanged mapping")
    verified = case["case_preparation"]["verified_sources"]
    source_hashes = owner.get("source_sha256", {})
    if (source_hashes.get("observation_archive") != verified["observations"]["sha256"]
            or source_hashes.get("case_manifest") != _sha_file(run / "CASE_MANIFEST.json")):
        raise ValueError("nearest-owner audit is bound to a different input or case manifest")
    producer = HERE / "audit_ko_nearest_owners.py"
    if not producer.is_file() or _sha_file(producer) != source_hashes.get("producer"):
        raise ValueError("nearest-owner audit producer bytes differ from its saved report")
    if owner.get("distance_comparison_is_not_a_bit_parity_gate") is not True:
        raise ValueError("nearest-owner audit does not distinguish ownership from distance rounding")
    return owner


def _check_coverage_figure_v1_history(run: Path, run_id: str, case):
    folder = run / "COVERAGE_FIGURE"
    receipt_path = folder / "receipt.json"
    receipt = _json(receipt_path)
    verified = case["case_preparation"]["verified_sources"]
    ir105 = next(row for row in case["case_preparation"]["raw16_channel_inventory"]
                 if row["channel"] == "ir105")
    if (receipt.get("run_id") != run_id or receipt.get("native_centers") != NATIVE_COLUMNS
            or receipt.get("outside_pixel_edge_footprint") != 0
            or receipt.get("raw_KO_sha256") != ir105["sha256"]
            or receipt.get("observation_archive_sha256") != verified["observations"]["sha256"]
            or receipt.get("case_manifest_sha256") != _sha_file(run / "CASE_MANIFEST.json")
            or receipt.get("original_provider_navigation_certified") is not False
            or receipt.get("pixel_UTC_verified") is not False):
        raise ValueError("historical v1 coverage figure is not bound to the same full native grid")
    manifest_path = HERE / "AUDIT_EXECUTED_SOURCES_v1" / "manifest.json"
    manifest = _json(manifest_path)
    rows = [row for row in manifest.get("sources", []) if row.get("name") == "plot_case_coverage.py"]
    if len(rows) != 1 or rows[0].get("sha256") != receipt.get("producer_sha256"):
        raise ValueError("historical v1 coverage source is not preserved by its execution-source archive")
    source_copy = manifest_path.parent / "plot_case_coverage.py.txt"
    if not source_copy.is_file() or _sha_file(source_copy) != receipt["producer_sha256"]:
        raise ValueError("historical v1 coverage producer bytes are absent/changed")
    for name, expected in receipt.get("outputs_sha256", {}).items():
        path = folder / name
        if path.parent.resolve() != folder.resolve() or not path.is_file() or _sha_file(path) != expected:
            raise ValueError(f"historical v1 coverage artifact hash mismatch: {name}")
    if set(receipt.get("outputs_sha256", {})) != {"KO_NATIVE_DOMAIN.png", "KO_NATIVE_DOMAIN.pdf"}:
        raise ValueError("historical v1 coverage output inventory differs")
    return receipt


def _check_coverage_figure(run: Path, run_id: str, case, owner):
    folder = run / "COVERAGE_FIGURE_v2"
    receipt_path = folder / "receipt.json"
    receipt = _json(receipt_path)
    if (receipt.get("schema") != "kdm6.native_KO_coverage_figure.v2"
            or receipt.get("run_id") != run_id
            or receipt.get("scope") != "read-only geometry visualization; no model or RTTOV calls"
            or receipt.get("native_centers") != NATIVE_COLUMNS
            or receipt.get("outside_pixel_edge_footprint") != 0
            or receipt.get("original_provider_navigation_certified") is not False
            or receipt.get("pixel_UTC_verified") is not False):
        raise ValueError("KO/native coverage-figure receipt is incomplete or has excessive scope")
    if (receipt.get("case_manifest_sha256") != _sha_file(run / "CASE_MANIFEST.json")
            or receipt.get("observation_archive_sha256") != case["case_preparation"]["verified_sources"]["observations"]["sha256"]):
        raise ValueError("KO/native coverage figure is not bound to this case/input archive")
    ir105 = next((row for row in case["case_preparation"]["raw16_channel_inventory"]
                  if row["channel"] == "ir105"), None)
    if ir105 is None or receipt.get("raw_KO_sha256") != ir105["sha256"]:
        raise ValueError("coverage figure does not use the pinned raw IR105 file")
    producer = HERE / "plot_case_coverage.py"
    if not producer.is_file() or _sha_file(producer) != receipt.get("producer_sha256"):
        raise ValueError("coverage-figure producer bytes differ from the saved receipt")
    for name, expected in receipt.get("outputs_sha256", {}).items():
        path = folder / name
        if path.parent.resolve() != folder.resolve() or not path.is_file() or _sha_file(path) != expected:
            raise ValueError(f"coverage-figure output hash mismatch: {name}")
    if set(receipt.get("outputs_sha256", {})) != {"KO_NATIVE_DOMAIN.png", "KO_NATIVE_DOMAIN.pdf"}:
        raise ValueError("coverage-figure output set differs from PNG/PDF contract")
    residual = _finite_float(receipt.get("KO_center_projection_max_residual_m"), "KO center projection residual")
    if residual > 1e-6:
        raise ValueError("coverage-figure KO centers do not recover header grid centers")
    margins = receipt.get("minimum_center_to_pixel_edge_clearance_km", {})
    if set(margins) != {"west", "east", "south", "north"} or any(_finite_float(v, "coverage clearance") <= 0 for v in margins.values()):
        raise ValueError("coverage figure lacks positive center-to-pixel-edge clearances")
    owner_path = run / "KO_NEAREST_OWNER_AUDIT.json"
    owner_producer = HERE / "audit_ko_nearest_owners.py"
    if (owner.get("source_sha256", {}).get("observation_archive") != receipt.get("observation_archive_sha256")
            or owner.get("source_sha256", {}).get("case_manifest") != receipt.get("case_manifest_sha256")
            or receipt.get("nearest_owner_audit_sha256") != _sha_file(owner_path)):
        raise ValueError("coverage figure and nearest-owner audit bind different inputs")
    distance_diff = _finite_float(owner.get("max_abs_distance_difference_from_saved_m"),
                                  "spherical/LCC distance difference")
    if owner.get("saved_distance_dtype") != "float32" or distance_diff <= 0.0:
        raise ValueError("nearest-owner report fails to distinguish spherical distance from stored LCC distance")
    observation_path = Path(case["case_preparation"]["paths"]["observations"])
    preflight_path = observation_path.parent / "PREFLIGHT.json"
    preflight = _json(preflight_path)
    preparer_path = observation_path.parent / "code" / "prepare_ko7_inputs.py"
    preparer_hash = preflight.get("source_hashes", {}).get("preparer")
    if (preflight.get("input_bundle", {}).get("sha256") != receipt.get("observation_archive_sha256")
            or not preparer_hash or not preparer_path.is_file()
            or _sha_file(preparer_path) != preparer_hash):
        raise ValueError("PREFLIGHT/upstream input preparer does not bind this observation archive")
    preparer_lines = preparer_path.read_text().splitlines()
    distance_lines = [(i + 1, line.strip()) for i, line in enumerate(preparer_lines)
                      if "dist = np.hypot(x - nearest_x, y - nearest_y).astype(np.float32)" in line]
    if len(distance_lines) != 1:
        raise ValueError("pinned upstream preparer lacks one unique stored LCC distance formula")
    distance_line = distance_lines[0][0]
    excerpt_start = max(1, distance_line - 13)
    excerpt_end = min(len(preparer_lines), distance_line + 2)
    receipt["_folder"] = folder
    receipt["_owner_audit"] = owner
    receipt["_owner_audit_path"] = owner_path
    receipt["_owner_producer_path"] = owner_producer
    receipt["_producer_path"] = producer
    receipt["_distance_provenance"] = {
        "preflight_sha256": _sha_file(preflight_path),
        "preflight_input_bundle_sha256": preflight["input_bundle"]["sha256"],
        "upstream_preparer_source_id": "ko7_model_domain_20261014/code/prepare_ko7_inputs.py",
        "upstream_preparer_sha256": preparer_hash,
        "stored_LCC_distance_source_line": distance_line,
        "upstream_distance_source_excerpt": [
            {"line": line_no, "text": preparer_lines[line_no - 1]}
            for line_no in range(excerpt_start, excerpt_end + 1)
        ],
        "stored_LCC_distance_definition": "LCC projected nearest-center hypot, stored as binary32.",
    }
    return receipt


def _raw_source_validity(run: Path, baseline_entries):
    """Read actual source DQF/QC from each exact baseline NPZ receipt."""
    result = np.zeros((NATIVE_COLUMNS, len(CHANNELS)), dtype=np.bool_)
    seen = np.zeros(NATIVE_COLUMNS, dtype=np.bool_)
    for entry in baseline_entries:
        ci, start, stop = entry["chunk_index"], entry["start"], entry["stop"]
        receipt_path = run / "baseline" / f"chunk_{ci:05d}" / "receipt.json"
        receipt = _json(receipt_path)
        artifact = _inside(run, Path(receipt["artifact"]), f"baseline chunk {ci} source QC")
        if _sha_file(artifact) != receipt["artifact_sha256"]:
            raise ValueError(f"baseline chunk {ci} changed before source-quality extraction")
        with np.load(artifact, allow_pickle=False) as archive:
            missing = {"flat_index", "source_dqf", "source_qc"} - set(archive.files)
            if missing:
                raise ValueError(f"baseline chunk {ci} lacks source quality arrays {sorted(missing)}")
            flat = np.array(archive["flat_index"], copy=True)
            dqf = np.array(archive["source_dqf"], copy=True)
            qc = np.array(archive["source_qc"], copy=True)
        if (flat.shape != (stop-start,) or not np.array_equal(flat, np.arange(start, stop))
                or dqf.shape != (stop-start, 7) or dqf.dtype.kind not in "iu"
                or qc.shape != dqf.shape or qc.dtype.kind not in "iuf"
                or not np.isfinite(qc).all() or seen[start:stop].any()):
            raise ValueError(f"baseline chunk {ci} source quality arrays have invalid coverage/type")
        result[start:stop] = (dqf == 0) & (qc == 0)
        seen[start:stop] = True
    if not seen.all():
        raise ValueError("raw source quality extraction did not cover all native columns")
    return result


def _check_independent_audits(run: Path, run_id: str, case_sha: str, final_path: Path,
                              baseline, final, baseline_path: Path,
                              physics_path: Path, objective_path: Path):
    baseline_audit = _json(baseline_path)
    if (baseline_audit.get("source_run_id") != run_id
            or baseline_audit.get("case_manifest_sha256") != case_sha
            or baseline_audit.get("completed_chunks_audited") != 129
            or baseline_audit.get("columns_audited") != NATIVE_COLUMNS
            or baseline_audit.get("all_original_indices_present") is not True
            or len(baseline_audit.get("records", [])) != 129
            or baseline_audit.get("no_physical_calls") is not True
            or _finite_float(baseline_audit.get("max_dot_error"), "independent baseline dot") > 1e-10):
        raise ValueError("independent baseline audit does not certify complete matching receipts")

    physics = _json(physics_path)
    if (physics.get("schema") != "kdm6.native_satellite_applied_physics_audit.v1"
            or physics.get("run_id") != run_id
            or physics.get("source_hashes", {}).get("final_sha256") != _sha_file(final_path)):
        raise ValueError("applied-physics audit is not bound to this final run")
    expected_phases = 2 if final.get("accepted_trial") else 1
    phases = physics.get("phases")
    if not isinstance(phases, list) or len(phases) != expected_phases:
        raise ValueError("applied-physics audit phase count differs from final decision")
    for phase in phases:
        if (phase.get("blocks") != 129 or phase.get("budgeted_blocks") != 129
                or phase.get("all_replay_bit_gates_passed") is not True
                or phase.get("RTTOV_call_count") != 0):
            raise ValueError("applied-physics audit does not pass all saved full-domain rows")
    # This report deliberately does not require a closed physical budget: the
    # audit itself states precipitation was not captured.

    objective = _json(objective_path)
    if (objective.get("schema") != "kdm6.native_satellite_final_independent_audit.v1"
            or objective.get("status") != "COMPLETED_AUDITED"
            or objective.get("run_id") != run_id
            or objective.get("completion", {}).get("final_json_sha256") != _sha_file(final_path)
            or objective.get("grid", {}).get("columns") != NATIVE_COLUMNS
            or objective.get("grid", {}).get("chunks") != 129
            or objective.get("finite_difference_gate", {}).get("passed") is not True
            or objective.get("accepted_trial") != final.get("accepted_trial")):
        raise ValueError("independent objective audit does not certify this completed final run")
    return baseline_audit, physics, objective


def _required_array(archive, name, shape, dtype_kind=None, dtype=None, *, require_finite=True):
    if name not in archive:
        raise ValueError(f"assembled.npz lacks {name}")
    value = archive[name]
    if value.shape != shape:
        raise ValueError(f"assembled {name} shape {value.shape} != {shape}")
    if dtype is not None and value.dtype != np.dtype(dtype):
        raise TypeError(f"assembled {name} dtype {value.dtype} != {np.dtype(dtype)}")
    if dtype_kind is not None and value.dtype.kind not in dtype_kind:
        raise TypeError(f"assembled {name} dtype must be in {dtype_kind}")
    if require_finite and value.dtype.kind in "fiu" and not np.isfinite(value).all():
        raise ValueError(f"assembled {name} contains nonfinite values")
    return value


def _read_final_z(run: Path, final, assembled):
    if not final.get("accepted_trial"):
        return np.zeros((2, NATIVE_COLUMNS, NATIVE_LEVELS), dtype=np.float64)
    name = final["accepted_trial"]
    if not isinstance(name, str) or Path(name).name != name or not name.endswith(".json"):
        raise ValueError("accepted trial must be one root-level manifest filename")
    manifest_path = run / name
    manifest = _json(manifest_path)
    if manifest.get("acceptance", {}).get("accepted") is not True:
        raise ValueError("FINAL accepted trial does not record acceptance")
    chunks = manifest.get("chunks")
    if not isinstance(chunks, list) or len(chunks) != 129:
        raise ValueError("accepted trial manifest lacks all 129 chunks")
    z = np.empty((2, NATIVE_COLUMNS, NATIVE_LEVELS), dtype=np.float64)
    seen = np.zeros(NATIVE_COLUMNS, dtype=np.bool_)
    for entry in chunks:
        ci = entry.get("chunk_index")
        if type(ci) is not int or ci not in range(129):
            raise ValueError("accepted trial has invalid chunk index")
        artifact = _inside(run, Path(entry["artifact"]), f"{name} chunk artifact")
        if _sha_file(artifact) != entry.get("artifact_sha256"):
            raise ValueError(f"accepted trial chunk {ci} artifact SHA mismatch")
        with np.load(artifact, allow_pickle=False) as archive:
            if "z" not in archive:
                raise ValueError(f"accepted trial chunk {ci} lacks normalized control z")
            local = archive["z"]
            flat = archive["flat_index"]
        if local.shape != (2, len(flat), NATIVE_LEVELS) or local.dtype != np.float64:
            raise ValueError(f"accepted trial chunk {ci} z shape/dtype mismatch")
        start, stop = entry["start"], entry["stop"]
        if (flat.shape != (stop-start,) or not np.array_equal(flat, np.arange(start, stop))
                or seen[start:stop].any()):
            raise ValueError(f"accepted trial chunk {ci} has duplicate/misordered columns")
        z[:, start:stop] = local
        seen[start:stop] = True
    if not seen.all() or not np.isfinite(z).all() or np.max(np.abs(z)) > 1.0:
        raise ValueError("accepted normalized controls do not cover the bounded full grid")
    active = assembled["active"].astype(np.bool_)
    if np.any(z[~active] != 0.0):
        raise ValueError("accepted normalized controls are nonzero outside active T/Q degrees")
    return z


def _write_reason_jsonl_gz(path: Path, flat_index, reasons):
    with Path(path).open("xb") as raw:
        with gzip.GzipFile(fileobj=raw, mode="wb", mtime=0) as stream:
            for index, reason in zip(flat_index, reasons):

REASON_CATEGORIES = {
    0: "admitted (no row exclusion reason)",
    1: "raw NC negative at initial state",
    2: "raw NI negative at initial state",
    3: "raw source DQF/QC invalid",
    4: "native optical profile invalid",
    5: "RTTOV model quality nonzero",
    6: "no baseline-supported channel / other reason",
}


def _reason_category(reason, raw_valid_row, profile_valid, quality_row, mask_row):
    text = str(reason).lower()
    if "negative_initial_nc" in text or "negative_nc" in text:
        return 1
    if "negative_initial_ni" in text or "negative_ni" in text:
        return 2
    if not bool(np.all(raw_valid_row)):
        return 3
    if not bool(profile_valid) or "post_microphysics_profile_invalid" in text:
        return 4
    if np.any(np.asarray(quality_row) != 0):
        return 5
    if not bool(np.any(mask_row)) or "no_channel_passed" in text:
        return 6
    return 0


def _reason_category_counts(reasons, raw_valid, profile_valid, quality, mask):
    primary = np.empty(len(reasons), dtype=np.uint8)
    counts = {str(code): 0 for code in REASON_CATEGORIES}
    multi = {str(code): 0 for code in REASON_CATEGORIES}
    for i, reason in enumerate(reasons):
        text = str(reason).lower()
        primary[i] = _reason_category(reason, raw_valid[i], profile_valid[i], quality[i], mask[i])
        counts[str(int(primary[i]))] += 1
        flags = {
            0: not bool(text.strip()) or "profile_admitted" in text,
            1: "negative_initial_nc" in text or "negative_nc" in text,
            2: "negative_initial_ni" in text or "negative_ni" in text,
            3: not bool(np.all(raw_valid[i])),
            4: "post_microphysics_profile_invalid" in text or not bool(profile_valid[i]),
            5: "rttov_quality_nonzero" in text,
            6: ("no_channel_passed" in text or (not np.any(mask[i]) and not np.any(raw_valid[i] == 0))),
        }
        # Raw profile reasons can contain multiple diagnoses. These counts are
        # intentionally multi-label; the grid category above is a stable summary.
        for code, active in flags.items():
            if active:
                multi[str(code)] += 1
    return primary, counts, multi

def _receipt_pack(run: Path, run_id: str, baseline, final, baseline_audit, physics, objective):
    def rttov_receipt(value):
        if not isinstance(value, dict):
            return None
        keys = ("mode", "baseline_operator", "K_calls", "direct_calls", "K_recovered",
                "nprofiles", "channels", "bt_coordinate", "pressure_gradients",
                "source_flat_indices", "config_hashes", "config_hash", "driver_flags",
                "recovery")
        return {key: value[key] for key in keys if key in value}

    def chunks(rows, stage):
        out = []
        for row in rows:
            index = row["chunk_index"]
            receipt_path = run / stage / f"chunk_{index:05d}" / "receipt.json"
            out.append({
                key: row.get(key) for key in (
                    "chunk_index", "start", "stop", "columns", "binding", "Z_partial",
                    "observation_numerator", "dot", "support_preserved", "valid_observations",
                    "calls", "artifact_sha256",
                )
            } | {"RTTOV_receipt": rttov_receipt(row.get("RTTOV_receipt")),
                 "receipt_sha256": _sha_file(receipt_path)})
        return out
    baseline_json = _json(run / "BASELINE.json")
    baseline_chunks = chunks(baseline_json["chunks"], "baseline")
    fd = _json(run / "FD.json")
    fd_trials = []
    for trial in fd.get("trials", []):
        endpoints = []
        for entry in trial.get("endpoints", []):
            ci = entry["chunk_index"]
            # The saved FD manifest carries completed endpoint receipts. Keep
            # only their numeric/provenance identity; never export file paths.
            endpoint = {key: entry.get(key) for key in (
                "chunk_index", "start", "stop", "columns", "binding", "Z_partial",
                "observation_numerator", "support_preserved", "support_error",
                "valid_observations", "artifact_sha256",
            )}
            artifact = entry.get("artifact")
            if artifact:
                path = _inside(run, Path(artifact), f"FD chunk {ci} artifact")
                endpoint["artifact_sha256"] = _sha_file(path)
                relative_parts = path.relative_to(run).parts
                if relative_parts:
                    endpoint["endpoint_stage"] = relative_parts[0]
                receipt_path = path.parent.parent / "receipt.json"
                if receipt_path.is_file():
                    endpoint["receipt_sha256"] = _sha_file(receipt_path)
            endpoints.append(endpoint)
        fd_trials.append({key: trial.get(key) for key in (
            "epsilon", "AD", "FD", "tolerance_ratio", "eligible", "passed"
        )} | {"endpoints": endpoints})
    trial_summaries = []
    for name in final.get("trials", []):
        manifest_path = run / name
        m = _json(manifest_path)
        trial_summaries.append({
            "name": Path(name).name, "manifest_sha256": _sha_file(manifest_path),
            "alpha": m.get("alpha"), "objective_valid": m.get("objective_valid"),
            "objective": m.get("objective"), "acceptance": m.get("acceptance"),
            "controls_sha256": m.get("controls_sha256"),
        })
    return {
        "run_id": run_id,
        "baseline": {
            key: baseline_json.get(key) for key in (
                "schema", "full_domain", "processed_columns", "globalZ", "objective",
                "dot", "call_totals",
            )
        } | {"manifest_sha256": _sha_file(run / "BASELINE.json"), "chunks": baseline_chunks},
        "sparse_fd": {"manifest_sha256": _sha_file(run / "FD.json"),
                      "scope": fd.get("scope"), "passed": fd.get("passed"), "trials": fd_trials},
        "trials": trial_summaries,
        "final": {key: final.get(key) for key in (
            "status", "accepted_trial", "trials", "processed_columns", "blocks",
            "gradient_verification", "native_host_steps", "native_cycles", "completed_at_UTC",
        )} | {"FINAL_sha256": _sha_file(run / "FINAL.json")},
        "independent_audits": {
            "baseline": {
                "sha256": baseline_audit["_sha256"],
                "completed_chunks_audited": baseline_audit["completed_chunks_audited"],
                "columns_audited": baseline_audit["columns_audited"],
                "source_valid_observations": baseline_audit["source_valid_observations"],
                "supported_observations": baseline_audit["supported_observations"],
                "observation_numerator": baseline_audit["observation_numerator"],
                "max_dot_error": baseline_audit["max_dot_error"],
                "all_original_indices_present": baseline_audit["all_original_indices_present"],
            },
            "applied_physics": {
                "sha256": physics["_sha256"], "schema": physics["schema"],
                "phases": [{key: phase.get(key) for key in (
                    "phase", "blocks", "cpp_value_call_count", "RTTOV_call_count",
                    "all_replay_bit_gates_passed", "budgeted_blocks",
                )} for phase in physics["phases"]],
                "limits": physics.get("limits", {}),
            },
            "final_objective": {
                "sha256": objective["_sha256"], "schema": objective["schema"],
                "status": objective["status"], "run_id": objective["run_id"],
                "grid": objective["grid"], "baseline": objective["baseline"],
                "control": objective["control"],
                "finite_difference_gate": objective["finite_difference_gate"],
                "trials": objective["trials"], "accepted_trial": objective["accepted_trial"],
                "completion": objective["completion"], "limits": objective.get("limits", []),
            },
        },
    }


def _load_trial_outputs(run: Path, final, baseline_arrays, global_z):
    output = []
    n = NATIVE_COLUMNS
    c = len(CHANNELS)
    run_id = _json(run / "CASE_MANIFEST.json")["run_id"]
    fixed_mask = baseline_arrays["mask"].astype(bool)
    fixed_weights = baseline_arrays["weights"]
    fixed_source = baseline_arrays["source_eligible"].astype(bool)
    fixed_obs = baseline_arrays["obs_bt"]
    for filename in final.get("trials", []):
        if not isinstance(filename, str) or Path(filename).name != filename or not filename.endswith(".json"):
            raise ValueError("FINAL trial list must contain root-level manifest filenames")
        manifest_path = run / filename
        manifest = _json(manifest_path)
        stage = Path(filename).stem
        if (manifest.get("full_domain") is not True or manifest.get("globalZ") != global_z
                or len(manifest.get("chunks", [])) != 129):
            raise ValueError(f"trial {filename} does not declare a full frozen-support evaluation")
        bt = np.full((n, c), np.nan, dtype=np.float64)
        quality = np.full((n, c), -1, dtype=np.int32)
        profile_valid = np.zeros(n, dtype=np.bool_)
        reason_rows = [""] * n
        seen = np.zeros(n, dtype=np.bool_)
        support_preserved = True
        call_totals = {}
        chunk_receipts = []
        for index, row in enumerate(sorted(manifest["chunks"], key=lambda v: v.get("chunk_index", -1))):
            ci = row.get("chunk_index")
            if type(ci) is not int or ci != index:
                raise ValueError(f"trial {stage} chunks are missing/duplicated/out of order")
            start, stop = ci * 512, min((ci + 1) * 512, n)
            receipt_path = run / stage / f"chunk_{ci:05d}" / "receipt.json"
            receipt = _json(receipt_path)
            if receipt != row or row.get("start") != start or row.get("stop") != stop:
                raise ValueError(f"trial {stage} chunk {ci} receipt/range mismatch")
            baseline_receipt = _json(run / "baseline" / f"chunk_{ci:05d}" / "receipt.json")
            binding_data = {"run_id": run_id, "stage": stage, "chunk": ci,
                            "baseline_sha": baseline_receipt["artifact_sha256"],
                            "controls_sha": manifest.get("controls_sha256")}
            expected_binding = _sha_bytes(json.dumps(
                binding_data, sort_keys=True, allow_nan=False, separators=(",", ":")
            ).encode())
            if receipt.get("binding") != expected_binding:
                raise ValueError(f"trial {stage} chunk {ci} binding differs from its baseline/control")
            artifact = _inside(run, Path(receipt["artifact"]), f"trial {stage} chunk artifact")
            if _sha_file(artifact) != receipt.get("artifact_sha256"):
                raise ValueError(f"trial {stage} chunk {ci} artifact SHA mismatch")
            with np.load(artifact, allow_pickle=False) as archive:
                needed = ("flat_index", "bt", "quality", "profile_valid", "reasons",
                          "mask", "weights", "source_eligible", "obs_bt")
                missing = sorted(set(needed) - set(archive.files))
                if missing:
                    raise ValueError(f"trial {stage} chunk {ci} lacks {missing}")
                flat = np.array(archive["flat_index"], copy=True)
                local_bt = np.array(archive["bt"], copy=True)
                local_quality = np.array(archive["quality"], copy=True)
                local_valid = np.array(archive["profile_valid"], copy=True)
                local_reasons = np.array(archive["reasons"], copy=True)
                local_mask = np.array(archive["mask"], copy=True)
                local_weights = np.array(archive["weights"], copy=True)
                local_source = np.array(archive["source_eligible"], copy=True)
                local_obs = np.array(archive["obs_bt"], copy=True)
            size = stop - start
            if (flat.shape != (size,) or not np.array_equal(flat, np.arange(start, stop))
                    or local_bt.shape != (size, c) or local_bt.dtype != np.float64
                    or local_quality.shape != (size, c) or local_quality.dtype.kind not in "iu"
                    or local_valid.shape != (size,) or local_valid.dtype != np.bool_
                    or local_reasons.shape != (size,) or local_reasons.dtype.kind not in "US"
                    or local_mask.shape != (size, c) or local_weights.shape != (size, c)
                    or local_mask.dtype != np.bool_ or local_source.shape != (size, c)
                    or local_source.dtype != np.bool_
                    or local_obs.shape != (size, c) or local_obs.dtype != np.float64
                    or not np.array_equal(local_mask.astype(bool), fixed_mask[start:stop])
                    or not np.array_equal(local_weights, fixed_weights[start:stop])
                    or not np.array_equal(local_source.astype(bool), fixed_source[start:stop])
                    or not np.array_equal(local_obs, fixed_obs[start:stop])):
                raise ValueError(f"trial {stage} chunk {ci} output metadata differs from frozen baseline")
            lost = fixed_mask[start:stop] & ((local_quality != 0) | ~local_valid[:, None] | ~np.isfinite(local_bt))
            support_preserved &= not bool(lost.any())
            if manifest.get("objective_valid") and lost.any():
                raise ValueError(f"trial {stage} claims a valid objective after losing baseline support")
            if seen[start:stop].any():
                raise ValueError(f"trial {stage} duplicates native columns")
            bt[start:stop], quality[start:stop] = local_bt, local_quality
            profile_valid[start:stop] = local_valid
            reason_rows[start:stop] = [str(value) for value in local_reasons]
            seen[start:stop] = True
            for key, value in receipt.get("calls", {}).items():
                call_totals[key] = call_totals.get(key, 0) + int(value)
            chunk_receipts.append({
                "chunk_index": ci, "start": start, "stop": stop, "columns": size,
                "receipt_sha256": _sha_file(receipt_path),
                "artifact_sha256": receipt["artifact_sha256"],
                "support_preserved": receipt.get("support_preserved"),
                "observation_numerator": receipt.get("observation_numerator"),
                "calls": receipt.get("calls"),
            })
        if not seen.all():
            raise ValueError(f"trial {stage} does not cover all native columns")
        if bool(manifest.get("objective_valid")) != bool(support_preserved):
            raise ValueError(f"trial {stage} objective validity differs from frozen support preservation")
        output.append({
            "name": Path(filename).stem,
            "filename": filename,
            "manifest_sha256": _sha_file(manifest_path),
            "alpha": manifest.get("alpha"),
            "globalZ": manifest.get("globalZ"),
            "controls_sha256": manifest.get("controls_sha256"),
            "objective_valid": manifest.get("objective_valid"),
            "support_preserved": support_preserved,
            "objective": manifest.get("objective"),
            "acceptance": manifest.get("acceptance"),
            "profile_valid_count": int(profile_valid.sum()),
            "call_totals": call_totals,
            "chunks": chunk_receipts,
            "arrays": {"flat_index": np.arange(n, dtype=np.int32), "bt": bt,
                       "quality": quality, "profile_valid": profile_valid},
            "row_reasons": reason_rows,
        })
    return output


def _archive_execution_sources(stage: Path, context):
    """Copy exact final private producer bytes bound by their reports."""
    target = stage / "execution_sources"
    target.mkdir()
    root = HERE
    candidates = []
    prior_source_records = []
    prior_manifest_path = root / "AUDIT_EXECUTED_SOURCES_v1" / "manifest.json"
    if prior_manifest_path.is_file():
        prior = _json(prior_manifest_path)
        if prior.get("run_id") != context["scope"]["run_id"]:
            raise ValueError("prior executed-source archive is from a different run")
        prior_archive = prior_manifest_path.parent
        for row in prior.get("sources", []):
            name, digest = row.get("name"), row.get("sha256")
            source_copy = prior_archive / f"{name}.txt"
            if not isinstance(name, str) or not source_copy.is_file() or _sha_file(source_copy) != digest:
                raise ValueError(f"preserved executed source archive is missing/changed: {name}")
            reports = []
            for report in row.get("reports", []):
                rel = Path(report["report"])
                if rel.is_absolute() or ".." in rel.parts:
                    raise ValueError("prior audit source report reference is not a safe relative path")
                report_path = root / rel
                if not report_path.is_file() or _sha_file(report_path) != report.get("sha256"):
                    raise ValueError(f"prior executed-source report changed: {report['report']}")
                reports.append({"report": rel.as_posix(), "sha256": report["sha256"]})
            prior_source_records.append({"name": name, "sha256": digest, "reports": reports,
                                         "purpose": "preserved in the existing private v1 source archive; pilot/partial history only"})
            candidates.append({"name": name, "sha256": digest, "bytes": source_copy.read_bytes(),
                               "purpose": "exact source for preserved pilot/partial history",
                               "reports": reports,
                               "attestation": "AUDIT_EXECUTED_SOURCES_v1 source and report hashes"})
        prior_copy = target / "AUDIT_EXECUTED_SOURCES_v1_manifest.json"
        shutil.copyfile(prior_manifest_path, prior_copy)

    def add_current(name, source, expected, purpose, report_path=None, report_field=None):
        source = Path(source)
        if not source.is_file():
            raise FileNotFoundError(f"required executed-source file is absent: {name}")
        actual = _sha_file(source)
        if not expected or actual != expected:
            raise ValueError(f"{name} differs from exact source hash in its report")
        reports = []
        if report_path is not None:
            rp = Path(report_path).resolve(strict=True)
            if rp == context["run"] or context["run"] not in rp.parents:
                if rp.parent.resolve() != HERE.resolve():
                    raise ValueError("source-report reference lies outside evidence roots")
                report_name = rp.name
            else:
                report_name = rp.relative_to(context["run"]).as_posix()
                report_name = f"RUN_v4/{report_name}"
            reports.append({"report": report_name, "sha256": _sha_file(rp),
                            "source_hash_field": report_field})
        candidates.append({"name": name, "sha256": actual, "bytes": source.read_bytes(),
                           "purpose": purpose, "reports": reports,
                           "attestation": "saved report source hash" if report_path else "packer self-hash"})

    objective_hash = context["objective_audit"].get("audit_source_sha256")
    if not objective_hash:
        raise ValueError("final objective audit does not bind its own auditor script")
    add_current("audit_native_satellite_final_v1.py", root / "audit_native_satellite_final_v1.py",
                objective_hash, "FINAL independent objective audit",
                context["audit_inputs"][2], "audit_source_sha256")

    baseline_hash = context["baseline_audit"].get("auditor_sha256")
    if not baseline_hash:
        raise ValueError("independent baseline audit does not bind its own auditor script")
    add_current("independent_receipt_audit.py", root / "independent_receipt_audit.py",
                baseline_hash, "independent baseline receipt audit",
                context["audit_inputs"][0], "auditor_sha256")

    physics_hash = context["physics_audit"].get("source_hashes", {}).get("audit_producer_sha256")
    if not physics_hash:
        raise ValueError("applied-physics audit does not bind its own auditor script")
    add_current("audit_saved_case_physics.py", root / "audit_saved_case_physics.py",
                physics_hash, "saved applied-physics audit",
                context["audit_inputs"][1], "source_hashes.audit_producer_sha256")

    coverage_hash = context["coverage_figure"].get("producer_sha256")
    coverage_producer = context["coverage_figure"].get("_producer_path", root / "plot_case_coverage.py")
    add_current(Path(coverage_producer).name, coverage_producer, coverage_hash,
                "read-only KO/native geometry figure",
                context["run"] / "COVERAGE_FIGURE_v2" / "receipt.json", "producer_sha256")

    owner_hash = context["owner_audit"].get("source_sha256", {}).get("producer")
    add_current("audit_ko_nearest_owners.py", root / "audit_ko_nearest_owners.py",
                owner_hash, "independent full-domain nearest-pixel owner audit",
                context["run"] / "KO_NEAREST_OWNER_AUDIT.json", "source_sha256.producer")

    add_current("publish_run_v4.py", Path(__file__), _sha_file(Path(__file__)),
                "private publication packer")
    records = []
    destinations = {}
    for item in candidates:
        name, digest, payload = item["name"], item["sha256"], item["bytes"]
        key = (name, digest)
        if key not in destinations:
            same_name = [old for old in destinations if old[0] == name]
            filename = f"{name}.txt" if not same_name else f"{Path(name).stem}__{digest[:8]}{Path(name).suffix}.txt"
            destination = target / filename
            with destination.open("xb") as stream:
                stream.write(payload)
            if _sha_file(destination) != digest:
                raise ValueError(f"archived source bytes changed while copying: {name}")
            destinations[key] = {"file": filename, "sha256": digest, "bytes": len(payload),
                                 "purposes": [], "reports": [], "attestations": []}
        record = destinations[key]
        record["purposes"].append(item["purpose"])
        record["reports"].extend(item["reports"])
        record["attestations"].append(item["attestation"])
    records = list(destinations.values())
    _write_json(stage / "execution_sources.json", {
        "sources": records,
        "prior_partial_source_manifest_sha256": _sha_file(prior_manifest_path) if prior_manifest_path.is_file() else None,
        "prior_partial_sources": prior_source_records,
        "scope": "Exact current-final and preserved pilot/partial producer/auditor bytes are copied only when report hashes attest those source SHA values. The full KDM/RTTOV implementation source remains identified by the source_v3 hash manifest.",
    })
    return records

def _redacted_input_audits(run: Path, native, raw, owner):
    # These audits are path-light, but redact path-like fields defensively.
    path_keys = {"path", "source_file", "source_path", "run_root", "root", "directory"}
    def scrub(value):
        if isinstance(value, dict):
            return {k: scrub(v) for k, v in value.items()
                    if k not in path_keys and k not in ("case_manifest_path",)}
        if isinstance(value, list):
            return [scrub(v) for v in value]
        if isinstance(value, str) and (value.startswith("/Users/") or value.startswith("/private/")):
            return "[private absolute path redacted]"
        return value
    pack = {
        "all_native_input_blocks": {
            "sha256": _sha_file(run / "ALL_NATIVE_INPUT_BLOCKS_AUDIT.json"),
            "report": native,
        },
        "raw16_quality_and_footprint": {
            "sha256": _sha_file(run / "RAW16_QUALITY_AND_FOOTPRINT_AUDIT.json"),
            "report": raw,
        },
        "nearest_owner_identity": {
            "sha256": _sha_file(run / "KO_NEAREST_OWNER_AUDIT.json"),
            "report": owner,
        },
    }
    return scrub(pack)


def _prepare(run: Path, out: Path, baseline_audit_path: Path,
             physics_audit_path: Path, objective_audit_path: Path):
    run = Path(run).resolve(strict=True)
    out = Path(out).absolute()
    if out.exists() or out.is_symlink():
        raise FileExistsError(f"publication output must be new: {out}")
    if any(parent.is_symlink() for parent in out.parents if parent.exists()):
        raise ValueError("publication output path must not traverse a symlink")
    if run != RUN_DEFAULT.resolve(strict=True):
        raise ValueError("only the source-pinned RUN_v4 directory is supported")
    if not (run / "FINAL.json").is_file():
        raise FileNotFoundError("publication preflight requires a completed RUN_v4/FINAL.json")
    case_path = run / "CASE_MANIFEST.json"
    case = _json(case_path)
    run_id = case.get("run_id")
    if not isinstance(run_id, str) or not run_id:
        raise ValueError("CASE_MANIFEST lacks run_id")
    if case.get("raw_host_number_units_certified") is not False:
        raise ValueError("RUN_v4 number-unit uncertainty must remain explicitly uncertified")
    baseline, baseline_entries = _check_baseline(run, run_id, _sha_file(case_path))
    final, summary, summary_dir, summary_files, final_path, summary_receipt = _check_final(
        run, run_id, baseline
    )
    native_audit, raw_audit = _check_input_audits(run, run_id, case)
    owner_audit = _check_owner_audit(run, run_id, case)
    coverage_figure = _check_coverage_figure(run, run_id, case, owner_audit)
    coverage_figure_v1 = _check_coverage_figure_v1_history(run, run_id, case)
    raw_source_valid = _raw_source_validity(run, baseline_entries)

    baseline_audit_path = _inside(run, baseline_audit_path, "independent baseline audit")
    physics_audit_path = _inside(run, physics_audit_path, "applied physics audit")
    objective_audit_path = Path(objective_audit_path)
    if objective_audit_path.is_symlink():
        raise ValueError("independent objective audit must not be a symlink")
    objective_audit_path = objective_audit_path.resolve(strict=True)
    if (objective_audit_path.parent.resolve() != HERE.resolve()
            and run not in objective_audit_path.parents):
        raise ValueError("independent objective audit must be in evidence root or RUN_v4")
    baseline_audit, physics, objective = _check_independent_audits(
        run, run_id, _sha_file(case_path), final_path, baseline, final,
        baseline_audit_path, physics_audit_path, objective_audit_path,
    )
    baseline_audit["_sha256"] = _sha_file(baseline_audit_path)
    physics["_sha256"] = _sha_file(physics_audit_path)
    objective["_sha256"] = _sha_file(objective_audit_path)

    with np.load(summary_dir / "assembled.npz", allow_pickle=False) as archive:
        b = NATIVE_COLUMNS
        l = NATIVE_LEVELS
        c = len(CHANNELS)
        a = {
            "flat_index": _required_array(archive, "flat_index", (b,), dtype_kind="iu"),
            "native_i_1based": _required_array(archive, "native_i_1based", (b,), dtype_kind="iu"),
            "native_j_1based": _required_array(archive, "native_j_1based", (b,), dtype_kind="iu"),
            "obs_bt": _required_array(archive, "obs_bt", (b, c), dtype=np.float64),
            "baseline_bt": _required_array(archive, "baseline_bt", (b, c), dtype=np.float64,
                                           require_finite=False),
            "final_bt": _required_array(archive, "final_bt", (b, c), dtype=np.float64,
                                         require_finite=False),
            "baseline_quality": _required_array(archive, "baseline_quality", (b, c), dtype_kind="iu"),
            "final_quality": _required_array(archive, "final_quality", (b, c), dtype_kind="iu"),
            "mask": _required_array(archive, "mask", (b, c), dtype_kind="biu"),
            "weights": _required_array(archive, "weights", (b, c), dtype=np.float64),
            "source_eligible": _required_array(archive, "source_eligible", (b, c), dtype_kind="biu"),
            "profile_valid": _required_array(archive, "profile_valid", (b,), dtype_kind="biu"),
            "final_profile_valid": _required_array(archive, "final_profile_valid", (b,), dtype_kind="biu"),
            "reasons": _required_array(archive, "reasons", (b,), dtype_kind="US"),
            "theta_m": _required_array(archive, "theta_m", (b, l), dtype=np.float64),
            "qv": _required_array(archive, "qv", (b, l), dtype=np.float64),
            "active": _required_array(archive, "active", (2, b, l), dtype_kind="biu"),
            "grad_theta_cost": _required_array(archive, "grad_theta_cost", (b, l), dtype=np.float64),
            "grad_qv_cost": _required_array(archive, "grad_qv_cost", (b, l), dtype=np.float64),
            "direction_theta": _required_array(archive, "direction_theta", (b, l), dtype=np.float64),
            "direction_qv": _required_array(archive, "direction_qv", (b, l), dtype=np.float64),
            "bt_jvp": _required_array(archive, "bt_jvp", (b, c), dtype=np.float64),
            "delta_theta": _required_array(archive, "delta_theta", (b, l), dtype=np.float64),
            "delta_qv": _required_array(archive, "delta_qv", (b, l), dtype=np.float64),
        }
        # Copy arrays out of the zip before it closes; no field is rounded or quantized.
        a = {key: np.array(value, copy=True) for key, value in a.items()}
    if not np.array_equal(a["flat_index"], np.arange(NATIVE_COLUMNS, dtype=a["flat_index"].dtype)):
        raise ValueError("assembled grid is not the complete j-major native index order")
    if not np.array_equal(a["native_i_1based"], a["flat_index"] % 234 + 1):
        raise ValueError("assembled native i coordinates disagree with flat indices")
    if not np.array_equal(a["native_j_1based"], a["flat_index"] // 234 + 1):
        raise ValueError("assembled native j coordinates disagree with flat indices")
    if not np.array_equal(a["mask"].astype(bool), a["source_eligible"].astype(bool) & (a["baseline_quality"] == 0)):
        raise ValueError("assembled frozen support differs from source/model quality gates")
    fixed_support = a["mask"].astype(bool)
    if (not np.isfinite(a["obs_bt"][fixed_support]).all()
            or not np.isfinite(a["baseline_bt"][fixed_support]).all()
            or not np.isfinite(a["final_bt"][fixed_support]).all()):
        raise ValueError("baseline/final BT must be finite on frozen support")
    expected_eligible = raw_source_valid & a["profile_valid"].astype(bool)[:, None]
    if not np.array_equal(a["source_eligible"].astype(bool), expected_eligible):
        raise ValueError("assembled source_eligible does not equal raw DQF/QC plus profile validity")
    if not np.isfinite(a["weights"]).all() or np.any(a["weights"][~a["mask"].astype(bool)] != 0.0):
        raise ValueError("assembled support weights are invalid")
    z = _read_final_z(run, final, a)

    source_snapshot_path = HERE / "SOURCE_SNAPSHOT_v3.json"
    source_manifest = _source_manifest(case, source_snapshot_path, case_path)
    reason_codes, reason_primary_counts, reason_multilabel_counts = _reason_category_counts(
        a["reasons"], raw_source_valid, a["profile_valid"].astype(bool),
        a["baseline_quality"], a["mask"].astype(bool),
    )
    trial_outputs = _load_trial_outputs(run, final, a, baseline["globalZ"])
    request = case["request"]
    request_path = HERE / "CASE_REQUEST_v4.json"
    if request != _json(request_path):
        raise ValueError("CASE_REQUEST_v4 differs from the run-bound request")
    scope = {
        "run_id": run_id,
        "model_snapshot_time": request["settings"]["model_snapshot_time"],
        "KO_nominal_slot": request["settings"]["KO_nominal_slot"],
        "grid": request["settings"]["grid"],
        "channels_1based": list(CHANNELS),
        "channel_names": list(CHANNEL_NAMES),
        "physics": request["settings"]["physics"],
        "local_physics_dt_seconds": request["settings"]["local_physics_dt_seconds"],
        "theta_m_scale_K": request["settings"]["theta_m_scale_K"],
        "qv_relative_scale": request["settings"]["qv_relative_scale"],
        "regularization_lambda": request["settings"]["regularization_lambda"],
        "huber_delta": request["settings"]["huber_delta"],
        "observation_sigma_K": request["settings"]["observation_sigma_K"],
        "bias_K": request["settings"]["bias_K"],
        "trial_alphas": request["settings"]["trial_alphas"],
        "fd_epsilons": request["fd_epsilons"],
        "model_quality_required": request["settings"]["model_quality_required"],
        "fixed_baseline_support": request["settings"]["fixed_baseline_support"],
        "native_host_steps": request["settings"]["native_host_steps"],
        "host_timestep_or_cycle": False,
        "raw_host_number_units_certified": case.get("raw_host_number_units_certified") is True,
        "optical_number_coordinate": case.get("optical_number_coordinate"),
        "physical_number_admission": False,
        "control_increment_definition": "delta_theta_m and delta_qv are accepted input control increments x_final-x_b; all zero because accepted_trial is None. They are distinct from KDM microphysics before/after state changes and tendencies.",
        "pixel_UTC_verified": request.get("source_scope", "").find("Pixel UTC unverified") < 0,
        "provider_original_navigation_certified": False,
        "fixed_upper_reference_layers": 27,
        "upper_reference_note": "26 fixed complete layers plus one live partial layer at native model top; trace gases are held from the pinned RTTOV template.",
        "scope_limit": "Offline full-domain first-order T/Q control case; no host step/cycle, no external data, no water/energy/number budget closure claim.",
    }

    baseline_receipts = _receipt_pack(
        run, run_id, baseline, final, baseline_audit, physics, objective
    )
    input_audits = _redacted_input_audits(run, native_audit, raw_audit, owner_audit)
    history = _json_safe_history()
    return {
        "run": run, "out": out, "final": final, "summary": summary,
        "summary_dir": summary_dir, "summary_files": summary_files,
        "arrays": a, "z": z, "source_manifest": source_manifest,

        "owner_audit": owner_audit,
        "reason_codes": reason_codes,
        "reason_primary_counts": reason_primary_counts,
        "reason_multilabel_counts": reason_multilabel_counts,
        "trial_outputs": trial_outputs,
        "scope": scope, "receipt_pack": baseline_receipts,
        "input_audits": input_audits, "history": history,
        "coverage_figure": coverage_figure,
        "coverage_figure_v1": coverage_figure_v1,
        "baseline_audit": baseline_audit, "physics_audit": physics,
        "objective_audit": objective,
        "audit_inputs": (baseline_audit_path, physics_audit_path, objective_audit_path),
    }


def _write_pack(context):
    out = context["out"]
    out.parent.mkdir(parents=True, exist_ok=True)
    if out.exists() or out.is_symlink():
        raise FileExistsError(f"publication pack output already exists: {out}")
    stage = Path(tempfile.mkdtemp(prefix=".run_v4_pack_", dir=out.parent))
    try:
        a = context["arrays"]
        idx = a["flat_index"]

                  baseline_quality=a["baseline_quality"], final_quality=a["final_quality"])

                  raw_source_valid=context["raw_source_valid"],

                  profile_valid=a["profile_valid"], final_profile_valid=a["final_profile_valid"],
                  active=a["active"])
        _write_reason_jsonl_gz(stage / "observation_reasons.jsonl.gz", idx, a["reasons"])
        _save_npz(stage / "native_reason_categories.npz", flat_index=idx,
                  primary_category_code=context["reason_codes"])
        _save_npz(stage / "native_state_controls.npz", flat_index=idx,
                  theta_m=a["theta_m"], qv=a["qv"])
        _save_npz(stage / "baseline_native_gradient_cost.npz", flat_index=idx,
                  grad_theta_cost=a["grad_theta_cost"], grad_qv_cost=a["grad_qv_cost"])
        _save_npz(stage / "baseline_native_tangent.npz", flat_index=idx,
                  direction_theta=a["direction_theta"], direction_qv=a["direction_qv"],
                  bt_jvp=a["bt_jvp"])
        _save_npz(stage / "native_control_delta.npz", flat_index=idx,
                  delta_theta=a["delta_theta"], delta_qv=a["delta_qv"])
        _save_npz(stage / "native_control_z.npz", flat_index=idx, z=context["z"])

        trial_dir = stage / "trials"
        trial_dir.mkdir()
        trial_index = []
        for trial in context["trial_outputs"]:
            arrays = trial.pop("arrays")
            row_reasons = trial.pop("row_reasons")
            npz_path = trial_dir / f"{trial['name']}_observations.npz"
            _save_npz(npz_path, **arrays)
            reason_path = trial_dir / f"{trial['name']}_reasons.jsonl.gz"
            _write_reason_jsonl_gz(reason_path, arrays["flat_index"], row_reasons)
            trial["arrays"] = {"file": npz_path.name, "sha256": _sha_file(npz_path),
                               "exact_reasons_file": reason_path.name,
                               "exact_reasons_sha256": _sha_file(reason_path)}
            trial_index.append(trial)
        _write_json(trial_dir / "trials.json", trial_index)

        # The numerical summary itself contains no machine paths; retain its
        # source-supported statistics without expanding private K/profile arrays.
        _write_json(stage / "summary.json", context["summary"])
        _write_json(stage / "scope.json", context["scope"])
        _write_json(stage / "source_manifest.json", context["source_manifest"])
        _write_json(stage / "execution_receipts.json", context["receipt_pack"])
        _write_json(stage / "input_audits.json", context["input_audits"])
        _write_json(stage / "history.json", context["history"])
        physics_report = {key: value for key, value in context["physics_audit"].items()
                          if key != "_sha256"}
        _write_json(stage / "applied_physics_audit.json", {
            "source_artifact_sha256": context["physics_audit"]["_sha256"],
            "redaction_note": "Absolute filesystem path strings are replaced; all saved numeric diagnostics, call counts, limits, and source hashes are retained verbatim.",
            "interpretation": {
                "replay_call_scope": "The baseline-only saved-state check uses 129 value-only C++ calls with zero RTTOV calls; this is a separate arithmetic replay, not the original KDM/RTTOV objective execution.",
                "column_totals": "water_column_totals_kg_m2 are per-column DEN*DELZ vertical sums followed by an unweighted sum across columns; they are not regional area-integrated mass.",

            "primary_category_codes": REASON_CATEGORIES,
            "primary_counts": context["reason_primary_counts"],

            "primary_precedence": [1, 2, 3, 4, 5, 6, 0],
            "exact_original_row_reasons": "UTF-8 strings are losslessly stored as one JSON object per native flat index in observation_reasons.jsonl.gz and per-trial *_reasons.jsonl.gz files.",

        pre_rt_count = (context["raw_source_valid"] & a["profile_valid"].astype(bool)[:, None]).sum(axis=1)

            "frozen_model_quality_supported_channel_count_range": [int(frozen_count.min()), int(frozen_count.max())],
            "raw_source_valid_observations": int(context["raw_source_valid"].sum()),
            "profile_valid_columns": int(a["profile_valid"].sum()),
            "baseline_supported_observations": int(a["mask"].sum()),
            "definition": "raw_source_valid=(saved per-block source_dqf==0 and source_qc==0); source_eligible=raw_source_valid AND profile_valid; frozen mask=source_eligible AND RTTOV model quality==0.",
        }
        _write_json(stage / "native_diagnostic_summaries.json", diagnostic_summary)
        _plot_coverage_maps(map_dir, a, context["raw_source_valid"])
        _plot_reason_grid(map_dir, context["reason_codes"], context["reason_primary_counts"])
        for name in context["summary_files"]:
            if name == "assembled.npz" or name == "summary.json":
                continue
            src = context["summary_dir"] / name
            if src.suffix.lower() not in (".png", ".pdf"):
                raise ValueError(f"unexpected summary output file type: {name}")
            shutil.copyfile(src, map_dir / name)
        coverage_dir = context["coverage_figure"]["_folder"]
        coverage_receipt = context["coverage_figure"]
        for name in ("KO_NATIVE_DOMAIN.png", "KO_NATIVE_DOMAIN.pdf"):
            shutil.copyfile(coverage_dir / name, map_dir / name)
        owner = context["owner_audit"]
        distance_provenance = coverage_receipt["_distance_provenance"]
        _write_json(stage / "geometry_coverage.json", {
            "scope": coverage_receipt["scope"],
            "figure_version": 2,
            "receipt_sha256": _sha_file(coverage_dir / "receipt.json"),
            "case_manifest_sha256": coverage_receipt["case_manifest_sha256"],
            "native_center_count": coverage_receipt["native_centers"],
            "native_centers_outside_actual_pixel_edge_footprint": coverage_receipt["outside_pixel_edge_footprint"],
            "center_to_pixel_edge_clearance_km": coverage_receipt["minimum_center_to_pixel_edge_clearance_km"],
            "KO_nearest_row_0based_minmax": coverage_receipt["KO_matched_row_minmax_0based"],
            "KO_nearest_col_0based_minmax": coverage_receipt["KO_matched_col_minmax_0based"],
            "max_nearest_pixel_LCC_distance_m_before_f32_storage": coverage_receipt["max_nearest_pixel_LCC_distance_m_before_f32_storage"],
            "max_stored_nearest_pixel_LCC_distance_m": coverage_receipt["max_stored_nearest_pixel_LCC_distance_m"],
            "stored_LCC_distance_f32_exact_matches": coverage_receipt["stored_LCC_distance_f32_exact_matches"],
            "max_nearest_pixel_spherical_distance_m": coverage_receipt["max_nearest_pixel_spherical_distance_m"],
            "spherical_vs_stored_LCC_max_abs_distance_difference_m": owner["max_abs_distance_difference_from_saved_m"],
            "distance_metric_contract": coverage_receipt["distance_metric_contract"],
            "KO_center_projection_max_residual_m": coverage_receipt["KO_center_projection_max_residual_m"],
            "spherical_LCC_radius_m_assumption": coverage_receipt["radius_m_assumption"],
            "nearest_owner_identity": {
                "audit_sha256": _sha_file(context["run"] / "KO_NEAREST_OWNER_AUDIT.json"),
                "producer_sha256": owner["source_sha256"]["producer"],
                "pixel_mismatches": owner["pixel_mismatches"],
                "unique_saved_pixel_owners": owner["unique_saved_pixel_owners"],
                "unique_recomputed_pixel_owners": owner["unique_recomputed_pixel_owners"],
                "direct_haversine_samples_all_match": all(row["matches_tree_and_saved"] for row in owner["direct_exhaustive_samples"]),
                "direct_haversine_sample_count": len(owner["direct_exhaustive_samples"]),
                "minimum_second_minus_first_owner_distance_m": owner["minimum_second_minus_first_distance_m"],
                "max_nearest_spherical_distance_m": owner["max_nearest_distance_m_assumed_radius"],
                "spherical_vs_stored_LCC_distance_difference_m_is_not_a_parity_gate": owner["max_abs_distance_difference_from_saved_m"],

            },
            "upstream_distance_definition_provenance": distance_provenance,
            "mapped_full_scene_BT_channel": {"name": "ir105", "ami_channel_id_1based": 13,
                                             "channel_slot_0based": CHANNEL_NAMES.index("ir105")},
            "raw_KO_sha256": coverage_receipt["raw_KO_sha256"],
            "observation_archive_sha256": coverage_receipt["observation_archive_sha256"],
            "upstream_preflight_distance_source_sha256": distance_provenance["upstream_preparer_sha256"],
            "producer_sha256": coverage_receipt["producer_sha256"],

            "pixel_UTC_verified": False,
            "execution_claim": "Geometry-only source/input coverage; not KDM or RTTOV execution evidence.",
        })
        v1_dir = context["run"] / "COVERAGE_FIGURE"
        v1_history = stage / "history" / "coverage_figure_v1"
        v1_history.mkdir(parents=True)
        v1_receipt = context["coverage_figure_v1"]
        for name in ("KO_NATIVE_DOMAIN.png", "KO_NATIVE_DOMAIN.pdf"):

            "uses a fixed category precedence; the exact per-row reason strings and multi-label counts are also retained.\n\n"
            "Native gradient and increment figures distinguish raw dJ/d-state from the scaled active-control "

        if out.exists() or out.is_symlink():
            raise FileExistsError(f"publication pack appeared during creation: {out}")
        os.rename(stage, out)
    except BaseException:
        shutil.rmtree(stage, ignore_errors=True)
        raise
    return out


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--run", type=Path, default=RUN_DEFAULT)
    parser.add_argument("--output", type=Path, default=OUT_DEFAULT)
    parser.add_argument("--baseline-audit", type=Path,
                        default=RUN_DEFAULT / "INDEPENDENT_BASELINE_FULL_AUDIT.json")
    parser.add_argument("--physics-audit", type=Path,
                        default=RUN_DEFAULT / "APPLIED_PHYSICS_AUDIT" / "ledger.json")
    parser.add_argument("--objective-audit", type=Path,
                        default=HERE / "FINAL_OBJECTIVE_AUDIT_v1.json")
    args = parser.parse_args(argv)
    context = _prepare(args.run, args.output, args.baseline_audit,
                       args.physics_audit, args.objective_audit)
    output = _write_pack(context)
    print(json.dumps({"output_directory": output.name,
                      "run_id": context["scope"]["run_id"],
                      "status": context["final"]["status"],
                      "artifact_count": len(json.loads((output / "publication_manifest.json").read_text())["artifacts"])}))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
