"""Independent saved-baseline arithmetic audit; no KDM/RTTOV/oracle imports."""
from pathlib import Path
import argparse
import hashlib
import io
import json
import math
from datetime import datetime, timezone

import numpy as np


def sha_bytes(data):
    return hashlib.sha256(data).hexdigest()


def json_id(value):
    data = json.dumps(value, sort_keys=True, allow_nan=False, separators=(",", ":")).encode()
    return sha_bytes(data)


def rounded_dot(left, right):
    products = np.asarray(left, dtype=np.float64) * np.asarray(right, dtype=np.float64)
    if not np.isfinite(products).all():
        raise ValueError("nonfinite saved directional product")
    return math.fsum(float(value) for value in products.ravel())


def audit(run, output):
    case_data = (run / "CASE_MANIFEST.json").read_bytes()
    case = json.loads(case_data)
    reference = case["case_preparation"]["packet_manifest"]["chunks"]
    records = []
    all_indices = []
    paths = sorted((run / "baseline").glob("chunk_*/receipt.json"))
    for path in paths:
        receipt_data = path.read_bytes()
        receipt = json.loads(receipt_data)
        index = receipt["chunk_index"]
        expected = reference[index]
        binding = json_id({"run_id": case["run_id"], "stage": "baseline", "chunk": index,
                           "baseline_sha": None, "controls_sha": None})
        assert receipt["binding"] == binding
        assert receipt["start"] == expected["start_flat"]
        assert receipt["stop"] == expected["stop_flat_exclusive"]
        assert receipt["columns"] == expected["columns"]
        artifact = Path(receipt["artifact"])
        assert artifact.parent.parent.resolve() == path.parent.resolve()
        encoded = artifact.read_bytes()
        assert sha_bytes(encoded) == receipt["artifact_sha256"]
        with np.load(io.BytesIO(encoded), allow_pickle=False) as archive:
            fields = {name: archive[name] for name in (
                "flat_index", "bt", "obs_bt", "source_dqf", "source_qc", "profile_valid",
                "quality", "source_eligible", "mask", "weights", "active", "qv",
                "grad_theta_numerator", "grad_qv_numerator", "direction_theta",
                "direction_qv", "bt_jvp")}
        flat = fields["flat_index"]
        assert np.array_equal(flat, np.arange(receipt["start"], receipt["stop"]))
        all_indices.extend(flat.tolist())
        source = (fields["source_dqf"] == 0) & (fields["source_qc"] == 0)
        eligible = source & fields["profile_valid"][:, None]
        supported = eligible & (fields["quality"] == 0)
        assert np.array_equal(eligible, fields["source_eligible"])
        assert np.array_equal(supported, fields["mask"])
        assert np.array_equal(fields["weights"], supported.astype(np.float64))
        assert int(supported.sum()) == receipt["valid_observations"]
        numerator_terms = []
        seed = np.zeros_like(fields["bt"])
        for row, channel in zip(*np.nonzero(supported)):
            residual = float(fields["bt"][row, channel]) - float(fields["obs_bt"][row, channel])
            assert math.isfinite(residual)
            if abs(residual) <= 1.0:
                term = (0.5 * residual) * residual
            else:
                term = abs(residual) - 0.5
            numerator_terms.append(term)
            seed[row, channel] = max(-1.0, min(1.0, residual))
        numerator = math.fsum(numerator_terms)
        assert numerator == receipt["observation_numerator"]
        z = float(np.count_nonzero(supported))
        assert z == receipt["Z_partial"]
        admitted = supported.any(axis=1)
        expected_active = np.zeros_like(fields["active"], dtype=bool)
        expected_active[0] = admitted[:, None]
        expected_active[1] = admitted[:, None] & (fields["qv"] > 0.0)
        assert np.array_equal(expected_active, fields["active"])
        lhs = rounded_dot(seed, fields["bt_jvp"])
        rhs = math.fsum((rounded_dot(fields["grad_theta_numerator"], fields["direction_theta"]),
                         rounded_dot(fields["grad_qv_numerator"], fields["direction_qv"])))
        assert lhs == receipt["dot"]["lhs"]
        assert rhs == receipt["dot"]["rhs"]
        error = abs(lhs-rhs) / max(abs(lhs), abs(rhs), 1e-12)
        assert error == receipt["dot"]["relative_error"]
        assert error <= 1e-10
        records.append({"chunk": index, "columns": int(flat.size), "source_valid": int(source.sum()),
                        "supported": int(supported.sum()), "numerator": numerator,
                        "dot_relative_error": error, "receipt_sha256": sha_bytes(receipt_data),
                        "artifact_sha256": receipt["artifact_sha256"]})
    assert len(set(all_indices)) == len(all_indices)
    report = {"scope": "completed baseline receipts listed at audit start; not full-case completion unless all129 are present",
              "source_run_id": case["run_id"], "case_manifest_sha256": sha_bytes(case_data),
              "completed_chunks_audited": len(records), "columns_audited": len(all_indices),
              "source_valid_observations": sum(row["source_valid"] for row in records),
              "supported_observations": sum(row["supported"] for row in records),
              "observation_numerator": math.fsum(row["numerator"] for row in records),
              "max_dot_error": max((row["dot_relative_error"] for row in records), default=None),
              "all_original_indices_present": sorted(all_indices) == list(range(65988)),
              "no_physical_calls": True, "records": records,
              "audited_at_UTC": datetime.now(timezone.utc).isoformat(),
              "auditor_sha256": sha_bytes(Path(__file__).read_bytes())}
    data = json.dumps(report, indent=2, allow_nan=False) + "\n"
    with output.open("x") as stream:
        stream.write(data)
    print(json.dumps({key: value for key, value in report.items() if key != "records"}), flush=True)


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--run", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    audit(args.run, args.output)
